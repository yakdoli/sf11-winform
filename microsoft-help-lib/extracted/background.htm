Background

 

You can control the background appearance of the chart area using Background property as follows.

 

[XAML]

 

<syncfusion:Chart>

  <syncfusion:ChartArea Background="LightBlue">

    <syncfusion:ChartSeries/>

  </syncfusion:ChartArea>

</syncfusion:Chart>

 

[C#]

 

Chart chart = new Chart();

ChartArea area = new ChartArea();

ChartSeries series = new ChartSeries();

 

//Set the Background properties

area.Background = new SolidColorBrush(Colors.LightBlue);

 

//Add Series in area

area.Series.Add(series);

 

//Add Area in Chart

chart.Areas.Add(area);
