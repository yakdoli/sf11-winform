Through Code

The Gallery View control can be created programmatically.

To create a Gallery View in ASP.NET Code:

1.  Create a new ASP.NET Web application.

2.  In the .cs file include the following directrives

 

[C#]

 

using  Syncfusion.Web.UI.WebControls.Shared;

using  Syncfusion.Web.UI;

 

[VB]

 

Imports  Syncfusion.Web.UI.WebControls.Shared

Imports  Syncfusion.Web.UI

 

3.   In code view, the control has been instantiated and added as follows-

 

[C#]    

 

private void Page_Init(object sender, EventArgs e)

{

        BuildGallery();

}

 

 

private void BuildGallery()

{

        GalleryView Gallery = new GalleryView();

        Gallery.ID = "Gallery1";

        Gallery.Width = 370;

        Gallery.DisplayMode = GalleryViewDisplayMode.GalleryView;

        Gallery.ItemsToShow = 4;

 

        GalleryViewItem GalleryItem = new GalleryViewItem();

        GalleryItem.Name = "Heading1";

        GalleryItem.ImageUrl = "Img/Heading1.png";

 

        GalleryViewItem GalleryItem1 = new GalleryViewItem();

        GalleryItem1.Name = "Heading2";

 

        GalleryItem1.ImageUrl = "Img/Heading2.png";

 

        GalleryViewItem GalleryItem2 = new GalleryViewItem();

        GalleryItem2.Name = "Heading3";

        GalleryItem2.ImageUrl = "Img/Heading3.png";

 

        GalleryViewItem GalleryItem3 = new GalleryViewItem();

        GalleryItem3.Name = "Intense";

        GalleryItem3.ImageUrl = "Img/Intense.png";

 

        GalleryViewItem GalleryItem4 = new GalleryViewItem();

        GalleryItem4.Name = "IntenseRef";

        GalleryItem4.ImageUrl = "Img/IntenseRef.png";

 

        GalleryViewItem GalleryItem5 = new GalleryViewItem();

        GalleryItem5.Name = "NoSpacing";

        GalleryItem5.ImageUrl = "Img/NoSpacing.png";

 

        GalleryViewItem GalleryItem6 = new GalleryViewItem();

        GalleryItem6.Name = "Title";

        GalleryItem6.ImageUrl = "Img/Title.png";

 

        GalleryViewExtraItem GalleryExtra = new GalleryViewExtraItem();

        GalleryExtra.Text = "Save Selection as new quick style";

        GalleryExtra.ImageUrl = "Img/Save16.png";

 

        GalleryViewExtraItem GalleryExtra1 = new GalleryViewExtraItem();

        GalleryExtra1.Text = "Save Selection as new quick style";

        GalleryExtra1.ImageUrl = "Img/ClearFormatting16.png";

 

        Gallery.Items.Add(GalleryItem);

        Gallery.Items.Add(GalleryItem1);

        Gallery.Items.Add(GalleryItem2);

        Gallery.Items.Add(GalleryItem3);

        Gallery.Items.Add(GalleryItem4);

        Gallery.Items.Add(GalleryItem5);

        Gallery.Items.Add(GalleryItem6);

 

        Gallery.ExtraItems.Add(GalleryExtra);

        Gallery.ExtraItems.Add(GalleryExtra1);

        form1.Controls.Add(Gallery);

}

 

 

[VB]

Protected Sub Page_Load(ByVal sender As Object, ByVal e As System.EventArgs) Handles Me.Load

        BuildGallery()

 End Sub
