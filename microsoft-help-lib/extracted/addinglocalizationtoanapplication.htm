Adding Localization to an application

1.   Create a folder called App_GlobalResources in the application folder and create your localization resource file (.resx) inside this folder

 

Figure 141: App_GlobalResources Folder

 

A part of the default English localization resource file is shown below:

Figure 142: Default English resource file

 

2.   To localize the static strings in the control, you can use the code snippets given below:

·      Specifying the culture using the Localize property-

[ASPX]

 

<syncfusion:gridgroupingcontrol enablecallbacks="false" id="GridGroupingControl2" runat="server" autoformat="Office 2007 Blue" datasourcecachingmode="ViewState" ShowSearchBox="true" >

<TableDescriptor AllowFilter="true" AllowNew="false" AllowEdit="false" Localize="fr-FR" >

</TableDescriptor>

</syncfusion:gridgroupingcontrol>

 

 

 

[C#]

 

    protected void Page_Load(object sender, EventArgs e)

    {

        if (!IsPostBack)

        {

            //Bind data to GridGroupingControl.

            GetData();

 

        }

        //Specify the Culture Code

        this.GridGroupingControl1.TableDescriptor.Localize = "fr-FR";

    }

 

 

[VB] 

 

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

            If (Not IsPostBack) Then

                  'Bind data to GridGroupingControl.

                  GetData()

 

            End If

            'Specify the Culture Code

            Me.GridGroupingControl1.TableDescriptor.Localize = "fr-FR"

End Sub

 

 

 

·      To get a specific resource file from your local resource file directory

 

[ASPX]

 

<syncfusion:gridgroupingcontrol enablecallbacks="false" id="GridGroupingControl2" runat="server" autoformat="Office 2007 Blue" datasourcecachingmode="ViewState" ShowSearchBox="true" >

<TableDescriptor AllowFilter="true" AllowNew="false" AllowEdit="false" Localize="fr-FR" LocalizationPath="~/App_LocalResources" >

</TableDescriptor>

</syncfusion:gridgroupingcontrol>

 

 

 

 

 [C#]

 

    protected void Page_Load(object sender, EventArgs e)

    {

        if (!IsPostBack)

        {

            //Bind data to GridGroupingControl.

            GetData();

 

        }

        //Specify the Culture Code

        this.GridGroupingControl1.TableDescriptor.Localize = "fr-FR";

        //Specify the resource folder

        this.GridGroupingControl1.TableDescriptor.LocalizationPath = "~/App_LocalResources";

    }

 

 

 

 

[VB]

 

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

            If (Not IsPostBack) Then

                  'Bind data to GridGroupingControl.

                  GetData()

 

            End If

            'Specify the Culture Code

            Me.GridGroupingControl1.TableDescriptor.Localize = "fr-FR"

            'Specify the resource folder

            Me.GridGroupingControl1.TableDescriptor.LocalizationPath = "~/App_LocalResources"

End Sub

 

 

·      Setting the CurrentUICulture property of the CurrentThread inside your Page_Load event:

 

[C#]

 

    protected void Page_Load(object sender, EventArgs e)

    {

        if (!IsPostBack)

        {

            //Bind data to GridGroupingControl.

            GetData();

 

        }

        //Specify the culture code

        System.Threading.Thread.CurrentThread.CurrentUICulture = new System.Globalization.CultureInfo("fr-FR");

 

    }

 

 

 

[VB]

 

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

            If (Not IsPostBack) Then

                  'Bind data to GridGroupingControl.

                  GetData()

 

            End If

            'Specify the culture code

            System.Threading.Thread.CurrentThread.CurrentUICulture = New System.Globalization.CultureInfo("fr-FR")

 

End Sub
