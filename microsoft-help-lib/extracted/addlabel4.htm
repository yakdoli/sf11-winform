Add Label

To add a text label, you have to create on object for the MapLabel and set the properties for the MapLabel.  Then add the created MapLabel object to ShapeFileLayer’s LabelCollections. This can be achieved in the following methods:

·      Through XAML

·      Through Code Behind

 

Through XAML

Following code illustrates how to add label through XAML:

 

[XAML]

 

<maps:MapControl LayeredContent="{Binding ElementName=shapeControl}"  x:Name="Map" NavigationControlPosition="Left" EnableColorPalette="True" ColorPalette="ColorPalette3" >

                        <maps:MapControl.Layers>

                            <maps:Layers>

                                <maps:ShapeFileLayer Uri="LabelsAndSymbolsDemo.ShapeFiles.world.shp"   x:Name="shapeControl" EnableLabel="{Binding IsChecked,ElementName=EnableLabel_ChkBox,Mode=TwoWay}" >

                                       

                                        <!-- Map Labels are added in Labels Collections with Latitude and Longitude Values, and with Customization Values -->

 

                                    <maps:ShapeFileLayer.LabelCollections>

                                       

                                        <maps:MapLabel Latitude="40" Longitude="-113" LabelText="North America" LabelBackground="Aqua" LabelDescription="North America" LabelFontFamily="Arial" LabelFontSize="20" LabelFontStyle="Italic"/>

                                        <maps:MapLabel Latitude="-17" Longitude="-71" LabelText="South America" LabelBackground="Black" LabelDescription="South America" LabelFontFamily="Times New Roman" LabelFontSize="20" LabelFontStyle="Italic" LabelForeground="Wheat"/>

                                        <maps:MapLabel Latitude="18" Longitude="-9" LabelText="Africa" LabelDescription="Africa" LabelFontFamily="Comic Sans MS" LabelFontSize="20" LabelFontStyle="Italic" LabelForeground="Blue" LabelBackground="BlanchedAlmond"/>

                                        <maps:MapLabel Latitude="51" Longitude="70" LabelText="Asia" LabelBackground="BurlyWood" LabelDescription="Asia" LabelFontFamily="Arial" LabelFontSize="20" LabelFontStyle="Italic"/>

                                        <maps:MapLabel Latitude="-25" Longitude="116" LabelText="Australia" LabelDescription="Australia" LabelFontFamily="Verdana" LabelFontSize="20" LabelFontStyle="Normal"/>

                                    </maps:ShapeFileLayer.LabelCollections>

                                   

                                </maps:ShapeFileLayer>

                            </maps:Layers>

                        </maps:MapControl.Layers>

                    </maps:MapControl>

 

Through Code Behind

Following code illustrates how to add label through code behind:

 

[C#]

 

// MapLabel added in Label Collection with Latitude, Longitude values, and with Customization values. Shape Control Denotes the Object of the ShapeFileLayer

 

 

this.shapeControl.LabelCollections.Add(

                new MapLabel

                {

                    LabelText = "North America",

                    Latitude = 40,

                    Longitude = -113,

                    LabelBackground = new SolidColorBrush(Colors.Aqua),

                    LabelFontFamily = new FontFamily("Arial"),

                    LabelFontStyle = FontStyles.Italic,

                    LabelForeground = new SolidColorBrush(Colors.Black),

                    LabelFontSize = 20

                });

 

            this.shapeControl.LabelCollections.Add(

                new MapLabel

                {

                    LabelText = "South America",

                    Latitude = -17,

                    Longitude = -71,

                    LabelBackground = new SolidColorBrush(Colors.Black),

                    LabelFontFamily = new FontFamily("Times New Roman"),

                    LabelFontStyle = FontStyles.Italic,

                    LabelForeground = new SolidColorBrush(Colors.Wheat),

                    LabelFontSize = 20

                });

 

            this.shapeControl.LabelCollections.Add(

                new MapLabel

                {

                    LabelText = "Africa",

                    Latitude = 18,

                    Longitude = -9,

                    LabelBackground = new SolidColorBrush(Colors.BlanchedAlmond),

                    LabelFontFamily = new FontFamily("Comic Sans MS"),

                    LabelFontStyle = FontStyles.Italic,

                    LabelForeground = new SolidColorBrush(Colors.Blue),

                    LabelFontSize = 20

                });

 

            this.shapeControl.LabelCollections.Add(

                new MapLabel

                {

                    LabelText = "Asia",

                    Latitude = 51,

                    Longitude = 70,

                    LabelBackground = new SolidColorBrush(Colors.BurlyWood),

                    LabelFontFamily = new FontFamily("Arial"),

                    LabelFontStyle = FontStyles.Italic,

                    LabelForeground = new SolidColorBrush(Colors.Black),

                    LabelFontSize = 20

                });

 

            this.shapeControl.LabelCollections.Add(

                new MapLabel

                {

                    LabelText = "Australia",

                    Latitude = -25,

                    Longitude = 116,

                    LabelBackground = new SolidColorBrush(Colors.Transparent),

                    LabelFontFamily = new FontFamily("Verdana"),

                    LabelFontStyle = FontStyles.Normal,

                    LabelForeground = new SolidColorBrush(Colors.Black),

                    LabelFontSize = 20

                });
