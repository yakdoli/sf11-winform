Adding Custom Skins to ASP.NET Grid

1.   Using the properties given in the Properties table, you can create a style sheet.

2.   Using this style sheet, you can customize the GridGroupingControl.

You can set the Custom CssClass using the code snippets given below-

[ASPX]

 

<%--GridGoupingControl--%>

 

<Syncfusion:GridGroupingControl ID="GridGroupingControl1" EnableCallbacks="False" runat="server" DataMember="DefaultView" DataSourceCachingMode="ViewState" BorderCollapse="Separate" CssClass="GridInsense" GroupDropAreaCssClass="GridInsenseGroupDropArea" DragSelectionBackColor="Yellow" HotTrackButtons="False"  >

<TableDescriptor AllowNew="False" Name="Customer Data" AllowFilter="True" TableOptions-SelectionBackColor="#E4DABA"

TableOptions-SelectionTextColor="black" AllowEdit="true" SortUpImage="../css/sortup.gif" SortDownImage="../css/sortdown.gif">

</TableDescriptor>

<Appearance>

<AnyRecordFieldCell CssClass="GridInsenseAnyRecord"/>
<AlternateRecordFieldCell CssClass="GridInsenseAlternateRecord"/>

<RecordPreviewCell CssClass="GridInsenseRecordPreview" />

<GroupIndentCell CssClass="GridInsenseGroupIndentCell" />

<GroupCaptionCell CssClass="GridInsenseGroupCaption" />

<AnySummaryCell CssClass="GridInsenseAnySummary" />

<GroupCaptionPlusMinusCell CssClass="GridInsensePlusMinus" />

<TopLeftHeaderCell CssClass="GridInsenseTopLeftHeaderCell" />

<RowHeaderCell CssClass="GridInsenseRowHeaders" />

<ColumnHeaderCell CssClass="GridInsenseColumnHeaders" />

<GroupHeaderRowHeaderCell CssClass="GridInsenseGroupedColumnHeaders" />

<FilterBarCell CssClass="GridInsenseFilterBarCell" />

</Appearance>

<TopLevelGroupOptions ShowFilterBar="true" FilterDialogCssClass="CustomCss" FilterDialogStyleSheet="../css/FilterStyle.css" FilterStatusBarCssClass="GridStatusBar" AllowExpressionFilter="true" ShowFilterBarTextCell="true"  AllowActiveFilteringMode="true" ShowFilterStatusMessage="true" FilterStatusBarWidth="450" />

</Syncfusion:GridGroupingControl>

 

 

<%--GridPager--%>

 

<syncfusion:GridPager ID="GridPager1" runat="server"  PagingControlID="GridGroupingControl1"  CssClass="SyncfusionPager" >

<PagerItems>

<syncfusion:NextPreviousGridPagerItem  ShowFirstPageButton="true" ShowPreviousPageButton="true"    />

<syncfusion:NumericGridPagerItem ButtonCount="7" />

<syncfusion:NextPreviousGridPagerItem ShowNextPageButton="true"  ShowLastPageButton="true" />                            

</PagerItems>

</syncfusion:GridPager>

 

 

[C#]

 

protected void Page_Load(object sender, EventArgs e)

    {

// CssClass for GridGroupingControl

 

        this.GridGroupingControl1.CssClass = "GridInsense";

 

        this.GridGroupingControl1.TableDescriptor.Appearance.AddNewRecordFieldCell.CssClass = "GridInsenseAnyRecord";

        this.GridGroupingControl1.TableDescriptor.Appearance.AlternateRecordFieldCell.CssClass = "GridInsenseAlternateRecord";

        this.GridGroupingControl1.TableDescriptor.Appearance.RecordPreviewCell.CssClass = "GridInsenseRecordPreview";

        this.GridGroupingControl1.TableDescriptor.Appearance.GroupIndentCell.CssClass = "GridInsenseGroupIndentCell";

        this.GridGroupingControl1.TableDescriptor.Appearance.GroupCaptionCell.CssClass = "GridInsenseGroupCaption";

        this.GridGroupingControl1.TableDescriptor.Appearance.AnySummaryCell.CssClass = "GridInsenseAnySummary";

        this.GridGroupingControl1.TableDescriptor.Appearance.GroupCaptionPlusMinusCell.CssClass = "GridInsensePlusMinus";

        this.GridGroupingControl1.TableDescriptor.Appearance.TopLeftHeaderCell.CssClass = "GridInsenseTopLeftHeaderCell";

        this.GridGroupingControl1.TableDescriptor.Appearance.RowHeaderCell.CssClass = "GridInsenseRowHeaders";

        this.GridGroupingControl1.TableDescriptor.Appearance.ColumnHeaderCell.CssClass = "GridInsenseColumnHeaders";

        this.GridGroupingControl1.TableDescriptor.Appearance.GroupHeaderRowHeaderCell.CssClass = "GridInsenseGroupedColumnHeaders";

        this.GridGroupingControl1.TableDescriptor.Appearance.FilterBarCell.CssClass = "GridInsenseFilterBarCell";

 

        this.GridGroupingControl1.TopLevelGroupOptions.FilterDialogCssClass = "CustomCss";

        this.GridGroupingControl1.TopLevelGroupOptions.FilterDialogStyleSheet = "../css/FilterStyle.css";

 

// CssClass for GridPager

        this.GridPager1.CssClass = "SyncfusionPager";

    }
