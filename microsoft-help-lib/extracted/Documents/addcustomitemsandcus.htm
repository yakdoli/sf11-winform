Add Custom Items and Custom Command to Ribbon Context Men

Custom ContextMenuItems also can be added to RibbonContextMenu with custom commands. The following code will explain the scenario.

C#

 

public MainPage()

{

      // Required to initialize variables

      InitializeComponent();

       myRibbon.OnRibbonContextMenuOpened += new RoutedEventHandler(myRibbon_OnRibbonContextMenuOpened);

}

 

ContextMenuItemAdv customItem;

 

void myRibbon_OnRibbonContextMenuOpened(object sender, RoutedEventArgs e)

{

       ContextMenuAdv adv = sender as ContextMenuAdv;

       if (customItem == null)

       {

           customItem = new ContextMenuItemAdv();

           customItem.Header = "Custom Command added by User";

           customItem.Command = CustomCommand;

           adv.Items.Add(customItem);

       }

}

 

public ICommand CustomCommand

{

      get

      {

          return new RibbonCommand(param => { MessageBox.Show("Custom Command Executing."); });

      }

}
