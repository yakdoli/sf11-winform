Binding to Object

To bind a collection to TreeViewAdv, use the ItemsSource property and define the Hierarchical Data Template that needs to display data from the collection.

1.   Create a class named ObjectModel.

 

[C#]

 

public class ObjectModel

    {

        public ObjectModel(string myString1, params ObjectModel[] myItems)

        {

            myString = myString1;

            ObservableCollection<ObjectModel> itemsObservableCollection = new ObservableCollection<ObjectModel>();

            foreach (var item in myItems)

            {

                itemsObservableCollection.Add(item);

            }

            Items = itemsObservableCollection;

        }

 

        public string myString { get; set; }

 

        public ObservableCollection<ObjectModel> Items { get; set; }

 

    }

 

2.   Create a class CustomItemsSource, which has a collection of business objects.

 

[C#]

 

public class CustomItemSource : ObservableCollection<ObjectModel>

    {

        public itemsource()

        {

            this.Add(new ObjectModel("Division 1", new ObjectModel("Director Fire & Life Safety", new ObjectModel("Asst Dr.Emergency Preparedness"), new ObjectModel("Fire Safety Specialist")), new ObjectModel("Director Security", new ObjectModel("Medical Security"), new ObjectModel("Campus Security")), new ObjectModel("Manager Training & Accreditation"), new ObjectModel("Director Police Operations", new ObjectModel("Staff Assistant"), new ObjectModel("Patrol Captain", new ObjectModel("Commander Sergeant", new ObjectModel("Police Officers"))), new ObjectModel("Asst Dr,Investigations", new ObjectModel("Directives Investigator"))), new ObjectModel("Director Operations")));

            this.Add(new ObjectModel("Division 2", new ObjectModel("Director Security"), new ObjectModel("Asst Director Security"), new ObjectModel("Asst Director Campus Security")));

            this.Add(new ObjectModel("Division 3", new ObjectModel("Teaching Professional", new ObjectModel("Co-ordinator"))));

        }

    }

 

3.   Bind the collection to the ItemsSource property of the TreeViewAdv.

[XAML]

 

<syncfusion:TreeViewAdv FullRowSelect="True" x:Name="treeview" Height="500" Width="400" BorderBrush="Black" BorderThickness="2">

            <!--Binding Collection to ItemsSource-->

            <syncfusion:TreeViewAdv.ItemsSource>

                <local:CustomItemSource/>

            </syncfusion:TreeViewAdv.ItemsSource>

            <!--HierarchicalDataTemplate-->

            <syncfusion:TreeViewAdv.ItemTemplate>

                <common:HierarchicalDataTemplate ItemsSource="{Binding Items}" >

                    <TextBlock Text="{Binding myString}" />

                </common:HierarchicalDataTemplate>

            </syncfusion:TreeViewAdv.ItemTemplate>

        </syncfusion:TreeViewAdv>
