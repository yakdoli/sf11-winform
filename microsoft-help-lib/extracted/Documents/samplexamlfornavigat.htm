Sample XAML for NavigationPopupStyle

The following XAML is used to customize the item-container style.

XAML

 

<SolidColorBrush x:Key="HierarchyNavigatorDropDownBorderBrushKey" Color="#FF727272"/>

<LinearGradientBrush x:Key="HierarchyNavigatorDropDownBackgroundBrushKey" EndPoint="0.5,1" StartPoint="0.5,0">

       <GradientStop Color="#FF86C6F4" Offset="0"/>

       <GradientStop Color="#FFC8DAE6" Offset="0.464"/>

       <GradientStop Color="#FF86C6F4" Offset="1"/>

</LinearGradientBrush>

<SolidColorBrush x:Key="HierarchyNavigatorPopupLeftBackgroundBrushKey" Color="#FFF1F1F1"/>

<SolidColorBrush x:Key="HierarchyNavigatorMenuItemSeparatorFillKey" Color="#FF052E4B"/>

<SolidColorBrush x:Key="HierarchyNavigatorMenuItemSeparatorStrokeKey" Color="#FF084875"/>

<Style x:Key="HierarchyNavigatorDropDownPopupStyle" TargetType="syncfusion:HierarchyNavigatorDropDownPopup">

       <Setter Property="Template">

              <Setter.Value>

                     <ControlTemplate TargetType="syncfusion:HierarchyNavigatorDropDownPopup">

                           <Grid x:Name="PART_Root" Opacity="{TemplateBinding Opacity}">

                                  <VisualStateManager.VisualStateGroups>

                                         <VisualStateGroup x:Name="CommonStates">

                                                <VisualState x:Name="Normal"/>

                                                <VisualState x:Name="Open">

                                                       <Storyboard>

                                                              <DoubleAnimationUsingKeyFrames BeginTime="00:00:00" Storyboard.TargetProperty="(UIElement.RenderTransform).(TransformGroup.Children)[0].(ScaleTransform.ScaleY)" Storyboard.TargetName="PART_PopUpBorder">

                                                                     <EasingDoubleKeyFrame KeyTime="00:00:00.3000000" Value="1"/>

                                                              </DoubleAnimationUsingKeyFrames>

                                                              <DoubleAnimationUsingKeyFrames BeginTime="00:00:00" Storyboard.TargetProperty="(UIElement.RenderTransform).(TransformGroup.Children)[0].(ScaleTransform.ScaleX)" Storyboard.TargetName="PART_PopUpBorder">

                                                                     <EasingDoubleKeyFrame KeyTime="00:00:00.3000000" Value="1"/>

                                                              </DoubleAnimationUsingKeyFrames>

                                                       </Storyboard>

                                                </VisualState>

                                                <VisualState x:Name="Close">

                                                       <Storyboard>

                                                              <DoubleAnimationUsingKeyFrames BeginTime="00:00:00" Storyboard.TargetProperty="(UIElement.RenderTransform).(TransformGroup.Children)[0].(ScaleTransform.ScaleY)" Storyboard.TargetName="PART_PopUpBorder">

                                                                     <EasingDoubleKeyFrame KeyTime="00:00:00.3000000" Value="0"/>

                                                              </DoubleAnimationUsingKeyFrames>

                                                              <DoubleAnimationUsingKeyFrames BeginTime="00:00:00" Storyboard.TargetProperty="(UIElement.RenderTransform).(TransformGroup.Children)[0].(ScaleTransform.ScaleX)" Storyboard.TargetName="PART_PopUpBorder">

                                                                     <EasingDoubleKeyFrame KeyTime="00:00:00.3000000" Value="0"/>

                                                              </DoubleAnimationUsingKeyFrames>

                                                       </Storyboard>

                                                </VisualState>

                                         </VisualStateGroup>

                                  </VisualStateManager.VisualStateGroups>

                                  <Popup x:Name="PART_DropDownItemsPopup" IsOpen="{TemplateBinding IsOpen}">

                                         <Border x:Name="PART_PopUpBorder" BorderBrush="{StaticResource HierarchyNavigatorDropDownBorderBrushKey}" BorderThickness="1" Background="{StaticResource HierarchyNavigatorDropDownBackgroundBrushKey}" Margin="0,0.4,0,0" MinWidth="220" Opacity="{TemplateBinding Opacity}">

                                                <Border.RenderTransform>

                                                       <TransformGroup>

                                                              <ScaleTransform ScaleY="0" ScaleX="0"/>

                                                              <SkewTransform/>

                                                              <RotateTransform/>

                                                              <TranslateTransform/>

                                                       </TransformGroup>

                                                </Border.RenderTransform>

                                                <Grid x:Name="PART_Container">

                                                       <Rectangle x:Name="PART_LeftBar" Fill="{StaticResource HierarchyNavigatorDropDownBackgroundBrushKey}" HorizontalAlignment="Left" RadiusY="2" RadiusX="2" Width="28"/>

                                                       <Rectangle x:Name="PART_ItemSeparatorFill" HorizontalAlignment="Left" Margin="27,0,0,0" Stroke="{StaticResource HierarchyNavigatorMenuItemSeparatorFillKey}" StrokeThickness="0.5" Width="1"/>

                                                       <Rectangle x:Name="PART_ItemSeparatorStroke" HorizontalAlignment="Left" Margin="28,0,0,0" Stroke="{StaticResource HierarchyNavigatorMenuItemSeparatorStrokeKey}" StrokeThickness="0.5" Width="1"/>

                                                       <ScrollViewer x:Name="PART_ScrollViewerRoot" BorderThickness="0" HorizontalScrollBarVisibility="Auto" MaxHeight="207" Padding="2" VerticalScrollBarVisibility="Auto">

                                                              <ItemsPresenter x:Name="PART_ItemsPresenter"/>

                                                       </ScrollViewer>

                                                </Grid>

                                         </Border>

                                  </Popup>

                           </Grid>

                     </ControlTemplate>

              </Setter.Value>

       </Setter>

</Style>

 

Each part of a template can be edited.
