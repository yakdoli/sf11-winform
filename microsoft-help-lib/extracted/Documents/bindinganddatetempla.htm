Binding and Date Template

Use ItemsSource property, to bind ContextMenuAdv to a collection of business objects. . The hierarchical data template allows you to bind data that is hierarchical and self-referencing.

[XAML]

 

<context:ContextMenuAdvService.ContextMenuAdv>

                <context:ContextMenuAdv>

                    <!--ItemSource Binding-->

                    <context:ContextMenuAdv.ItemsSource>

                        <local:CustomItemSource/>

                    </context:ContextMenuAdv.ItemsSource>

                    <!--Item Template-->

                    <context:ContextMenuAdv.ItemTemplate>

                        <common:HierarchicalDataTemplate ItemsSource="{Binding Items}">

                            <TextBlock Text="{Binding myString}"/>

                        </common:HierarchicalDataTemplate>

                    </context:ContextMenuAdv.ItemTemplate>

                </context:ContextMenuAdv>

    </context:ContextMenuAdvService.ContextMenuAdv>

 

The following is the Business Object used in this example.

[XAML]

 

    public class CustomItemSource : ObservableCollection<ObjectModel>

    {

        public CustomItemSource()

        {

            this.Add(new ObjectModel("Division 1",

                         new ObjectModel("Director Fire & Life Safety",

                             new ObjectModel("Asst Dr.Emergency Preparedness")),

                         new ObjectModel("Director Security",

                             new ObjectModel("Medical Security"),

                             new ObjectModel("Campus Security"))));

            this.Add(new ObjectModel("Division 2",

                         new ObjectModel("Director Security"),

                         new ObjectModel("Asst Director Security"),

                         new ObjectModel("Asst Director Campus Security")));

            this.Add(new ObjectModel("Division 3",

                         new ObjectModel("Teaching Professional",

                            new ObjectModel("Co-ordinator"))));

        }

    }
