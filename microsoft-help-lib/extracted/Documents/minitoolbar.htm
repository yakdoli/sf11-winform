Note: The horizontal offset and vertical offset property decides where the mini toolbar to display. Usually in Office Applications the mini toolbar showed when the text in the Editor selected. The scenario can be implemented as follows.

 

C#

 

Private void Editor_MouseMove(object sender, MouseEventArgs e)

{

     _mousepos = e.GetPosition(Editor);

     if (Editor.Selection.Text.Length == 0)

     {

          minitoolbar.IsOpen = false;

     }

}

 

Private void Editor_SelectionChanged(object sender, RoutedEventArgs e)

{

     if (minitoolbar != null)

     {

         minitoolbar.Target = Editor;

         minitoolbar.HorizontalOffset = _mousepos.X;

         minitoolbar.VerticalOffset = _mousepos.Y;

         minitoolbar.IsOpen = true;

     }

}
