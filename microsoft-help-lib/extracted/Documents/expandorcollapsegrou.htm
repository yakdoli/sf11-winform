Expand or Collapse GroupBar Item

 

You can expand or collapse the GroupBar Items by using the IsExpanded property. The following code example illustrates how to set this property.

 

[XAML]

 

<syncfusion:GroupBar x:Name="groupbar" Width="250" VisualMode="StackMode" Height="450" Orientation="Vertical" >

<syncfusion:GroupBarItem x:Name="groupbaritem" IsExpanded="True" Background="White" HeaderText="Calendar" >            

<Grid>

<TextBlock Text="TextBlock" />

</Grid>

</ syncfusion:GroupBarItem>

</syncfusion:GroupBar>

 

[C#]

 

GroupBar groupbar = new GroupBar();

groupbar.Width = 200;

groupbar.Height = 200;

groupbar.VisualMode = VisualMode.StackMode;

groupbar.Theme = VisualStyle.Office2007Blue;

groupbar.Orientation = Orientation.Vertical;

GroupBarItem groupbaritem = new GroupBarItem();

groupbaritem.HeaderText = "Groupbaritem";

Grid grid = new Grid();

TextBlock tb = new TextBlock();

tb.Text = "Content";

grid.Children.Add(tb);

groupbaritem.Content = grid;

groupbaritem.IsExpanded = true;

groupbar.Items.Add(groupbaritem);

LayoutRoot.Children.Add(groupbar);
