Header Aligning

            After adding the header, the Tab Item may align the header by Left, Right and Center. We can set this property in xaml and code behind also.

[Xaml]

  <syncfusion:TabControlAdv Name="MyTabCtrl" Height="300" Width="350"                

                                               TabScrollStyle="Extended">

            <syncfusion:TabItemAdv Header="TabItem1" HeaderAlignment="Center"  Width="100" />

            <syncfusion:TabItemAdv Header="TabItem2" HeaderAlignment="Left" Width="100" />

            <syncfusion:TabItemAdv Header="TabItem3" HeaderAlignment="Right" Width="100" />

  </syncfusion:TabControlAdv>

 

 

[C#]

TabControlAdv myTabCtrl = new TabControlAdv();

          TabItemAdv item1 = new TabItemAdv { Header = "TabItem1", HeaderAlignment = HeaderAlignment.Center , Width =100 };

            TabItemAdv item2 = new TabItemAdv { Header = "TabItem2", HeaderAlignment = HeaderAlignment.Left, Width =100 };

            TabItemAdv item3 = new TabItemAdv { Header = "TabItem3", HeaderAlignment = HeaderAlignment.Right , Width =100 };

            myTabCtrl.Items.Add(item1); myTabCtrl.Items.Add(item2); myTabCtrl.Items.Add(item3);

            this.LayoutRoot.Children.Add(myTabCtrl);
