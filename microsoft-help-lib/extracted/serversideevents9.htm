Server-Side Events

 

A CallbackMultiplexer is non-visual web control that lets you refresh one or more callback controls within the same callback event.

 

The server side event can be invoked to refresh the list of CallbackPanels on client side.

 

 

Event

 

Description

Callback

Server side event that is handled with the client side args on calling callback method of client side object.

 

Handle this event to change the contents of CallbackPanels according to the event args sent by the client.

 

For example, if you want to send custom callback event to the page, where you can either decide to refresh one or more callback panels. You can use ControlsToRefresh property of CallbackMultiplexer to add CallbackPanels to this list controls that should be refreshed.

 

The following code sample demonstrates how to refresh CallbackPanels.

 

[ASPX]

 

<table>

    <tr>

        <td>

            <ssw:CallbackPanel ID="CallbackPanel1" runat="server" Height="50px" Width="125px">

                <asp:Label ID="Label1" Text="CallbackPanel1" runat="server"></asp:Label>

            </ssw:CallbackPanel>

        </td>

        <td>

            <ssw:CallbackPanel ID="CallbackPanel2" runat="server" Height="50px" Width="125px">

                <asp:Label ID="Label2" Text="CallbackPanel2" runat="server"></asp:Label>

            </ssw:CallbackPanel>

        </td>

    </tr>

</table>

<asp:Button ID="Button1" runat="server" OnClientClick="_sfMultiplexer1.callback('Panel1'); return false;"

    Text="Refresh Panel1" />

<asp:Button ID="Button2" runat="server" OnClientClick="_sfMultiplexer1.callback('Panel2'); return false;"

    Text="Refresh Panel2" />

<asp:Button ID="Button3" runat="server" OnClientClick="_sfMultiplexer1.callback('BothPanels'); return false;"

    Text="Refresh both Panels" />

<ssw:CallbackMultiplexer ID="CallbackMultiplexer1" ClientObjectId="_sfMultiplexer1" runat="server" OnCallback="CallbackMultiplexer1_Callback" />

 

[C#]

 

protected void CallbackMultiplexer1_Callback(object sender, Syncfusion.Web.UI.WebControls.Shared.CallbackEventArgs e)

{

       switch( e.CallbackArgument )

       {                         

              case "Panel1":

                     CallbackPanel1.BackColor = GetRandomBackColor();

                     CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel1 );

                     break;

              case "Panel2":

                     CallbackPanel2.BackColor = GetRandomBackColor();

                     CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel2 );             

                     break;

              case "BothPanels":

                     CallbackPanel1.BackColor = GetRandomBackColor();

                     CallbackPanel2.BackColor = GetRandomBackColor();

                     CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel1 );

                     CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel2 );

                     break;

       }

}

 

private System.Drawing.Color GetRandomBackColor()

{

       System.Threading.Thread.Sleep( 1 );

       Random random = new Random();           

       int nRed = random.Next( 0, 255 );

       int nGreen = random.Next( 0, 255 );

       int nBlue = random.Next( 0, 255 );

       System.Drawing.Color color = System.Drawing.Color.FromArgb( nRed, nGreen, nBlue );

       return color;

}
