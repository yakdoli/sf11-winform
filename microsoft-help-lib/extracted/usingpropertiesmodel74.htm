Using Properties Model

The following steps, explains the appearance of Menu Control using the Properties model:

1.   In the Controller, create an instance for the MobMenuModel and pass the instance through ViewData to View as given below.

 

 

[Controller]

 

        publicActionResult Menu()

        {

            MobMenuModel model = newMobMenuModel()

            {

                AutoFormat = MobSkins.DarkNight,

                Items = newList<MenuItem>()

                {

                    newMenuItem(){ Text="FaceBook", ImageUrl="~/Content/Menu/Images/Facebook.png", ImageAttributes=new { @class = "menu-img" },

                    Items=newList<MenuItem>(){ newMenuItem(){ Text="Home Page" }, newMenuItem(){ Text="Log on" } }

                    },

                    newMenuItem(){ Text="Twitter", ImageUrl="~/Content/Menu/Images/twitter.png", ImageAttributes=new { @class = "menu-img" },

                    Items=newList<MenuItem>(){ newMenuItem(){ Text="Home Page" }}

                    },

                    newMenuItem(){ Text="Google", ImageUrl="~/Content/Menu/Images/Google.png", ImageAttributes=new { @class = "menu-img" },

                    Items=newList<MenuItem>(){ newMenuItem(){ Text="Home Page" }, newMenuItem(){ Text="Gmail" }, newMenuItem(){ Text="Orkut" },

                    newMenuItem(){ Text="Maps" }, newMenuItem(){ Text="News" }}

                    },

                    newMenuItem(){Text="YouTube", ImageUrl="~/Content/Menu/Images/youtube.png", ImageAttributes=new { @class = "menu-img" }},

                    newMenuItem(){Text="Books", ImageUrl="~/Content/Menu/Images/book.png", ImageAttributes=new { @class = "menu-img" }},

                    newMenuItem(){Text="Games", ImageUrl="~/Content/Menu/Images/game.png", ImageAttributes=new { @class = "menu-img" }}

                }

            };

            ViewData["ApperanceMenu"] = model;

            return View();

        }

 

 

2.   In View, invoke the menu helper with the ViewData key as the first argument.

 

[ASPX]

 

<% = Html.MobSyncfusion().Menu("ApperanceMenu")%>

 

 

 

[Razor]

 

@{ Html.MobSyncfusion().Menu("ApperanceMenu").Render();}

 

 

 

3.   Build and run the application.
