MultiRow Grid

 

MultiRow Records

 

A single record in a DataTable can span over multiple rows in a GridGroupingControl. This provides better readability and conserves horizontal space. Such multirow spans can be very easily and declaratively defined in the aspx.

 

[ASPX]

 

<TableDescriptor AllowEdit="False" AllowNew="False" AllowFilter="False">          

<ColumnSets>

    <sfwg:GridColumnSetDescriptor Name="ColumnSetId">                                           

<ColumnSpans>                                               

    <sfwg:GridColumnSpanDescriptor Range="R0C0:R1C0" Name="EmployeeID"></sfwg:GridColumnSpanDescriptor>

</ColumnSpans>

</sfwg:GridColumnSetDescriptor>

 

<sfwg:GridColumnSetDescriptor Name="ColumnSetPhoto">                                          

    <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0:R1C0" Name="SmallPhoto"></sfwg:GridColumnSpanDescriptor>

    </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

<sfwg:GridColumnSetDescriptor Name="ColumnSetName">

    <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0" Name="FirstName"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C0" Name="LastName"></sfwg:GridColumnSpanDescriptor>

     </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

<sfwg:GridColumnSetDescriptor Name="ColumnSetTitle">

     <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0" Name="Title"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C0" Name="TitleOfCourtesy"></sfwg:GridColumnSpanDescriptor>

     </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

<sfwg:GridColumnSetDescriptor Name="Phone">

    <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0:R0C1" Name="HomePhone"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R0C2" Name="Extension"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C0" Name="ReportsTo"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C1" Name="BirthDate"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C2" Name="HireDate"></sfwg:GridColumnSpanDescriptor>

 

    </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

 

<sfwg:GridColumnSetDescriptor Name="ColumnSetAddress">

    <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0:R0C1" Name="Address"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C0" Name="City"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C1" Name="PostalCode"></sfwg:GridColumnSpanDescriptor>

    </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

<sfwg:GridColumnSetDescriptor Name="ColumnSetRegionCountry">

    <ColumnSpans>

        <sfwg:GridColumnSpanDescriptor Range="R0C0" Name="Region"></sfwg:GridColumnSpanDescriptor>

        <sfwg:GridColumnSpanDescriptor Range="R1C0" Name="Country"></sfwg:GridColumnSpanDescriptor>

    </ColumnSpans>

</sfwg:GridColumnSetDescriptor>

</ColumnSets>

</TableDescriptor>
