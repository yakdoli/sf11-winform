The following steps describe how to use these client-side methods.

1.   In the view, create a ul-li hierarchy of tree-view nodes and invoke the TreeView helper with the control ID as the first argument and the tree-view content ID as the second argument.

 

View

<% Node node1 = new Node()

   {

       Name = "Node1",

       Label = "Circle",

       LabelHorizontalAlignment = Horizontal.Center,

       LabelVerticalAlignment = Vertical.Middle,

       OffsetX = 200,

       OffsetY = 100,

       Shape = Shapes.Ellipse,

       Height = 50.00,

       Width = 50.00,

       BackgroundColor = "orange",

       BorderColor = "black",

       BorderWidth = 1,

       LabelFontColor = "black",

       LabelFontSize = 12,

       LabelFontFamily="Times New Roman"

   };

   Node node2 = new Node()

   {

       Name = "Node2",

       Label = "Rectangle",

       LabelHorizontalAlignment = Horizontal.Center,

       LabelVerticalAlignment = Vertical.Middle,

       OffsetX = 200.00,

       OffsetY = 400.00,

       Shape = Shapes.Rectangle,

       Height = 50.00,

       Width = 100.00,

       BackgroundColor = "orange",

       BorderColor = "black",

       LabelBorderWidth = 1,

       LabelFontColor = "black",

       LabelFontSize = 12,

       LabelFontFamily="Times New Roman"

   };

   LineConnector line1 = new LineConnector()

   {

       Name="line1",

       HeadNode = node1,

       TailNode = node2,      

       ConnectorType = ConnectorType.Straight,

       LineColor = "black",

       LineWidth = 1

   };

 

%>

<%{

      Html.Syncfusion().Diagram("FlatDiagram")    

          .Nodes(nodes=>nodes.Add(node1).Add(node2))

          .Connectors(conn=>conn.Add(line1)) 

          .DiagramMode(DiagramMode.SVG)

         .Render();

  }

%>

 

2.   In JavaScript, use the methods as shown in the following code: 

 

[Javscript] 

<script type="text/javascript">

             function ChangeColor(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeBackgroundColor("Node1","Red");  

             }

             function ChangeBorderColor(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeBorderColor("Node1","Blue");

             }

             function ChangeFontColor(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeFontColor ("Node1","Yellow");       

             }

             function onWidthChange(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeWidth ("Node1",70);      

             }

             function onBorderWidthChange(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeBorderWidth ("Node1",3);      

             }

             function onFontSizeChange(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeFontSize ("Node1",16);      

              }

             function onHeightChange(sender, args) {

             diagram = $find(FlatDiagram);

             diagram.setNodeHeight ("Node1",70);    

              }

            function onOffsetXChange(sender, args) {

            diagram = $find(FlatDiagram);

            diagram.setNodeOffsetX ("Node1",300);    

            }

            function onOffsetYChange(sender, args) {

            diagram = $find(FlatDiagram);

            diagram.setNodeOffsetY ("Node1",200);      

            }
            function dynamicConnector(sender, args) {

            diagram = $find(FlatDiagram);

            diagram.setDynamicConnector("Orthogonal");

            }

      </script>
