UploadBox Server-side Events

 

These events can be handled on the server-side to execute the specified functions and perform the required task. The events supported by UploadBox are as follows.

 

Server-Side Event

Description

FileUploaded

Specifies the function to be executed after the file upload process is complete.

BeforeFileSave

To cancel the file saving on disk or save in separate folder.

FileExist

Specifies the function to be executed when the target folder already has a file with the same name. FileExists will be ignored, when AutoPostBackOnUpload is set to False.

 

Describe the FileUploaded, BeforeFileSave and FileExist events, and then define the task to be performed.

 

[ASPX]

 

<syncfusion:UploadBox id="UploadBox1" runat="server" OnBeforeFileSave="UploadBox1_BeforeFileSave" OnFileExists="UploadBox1_FileExists" OnFileUploaded="UploadBox1_FileUploaded">

</syncfusion:UploadBox>

 

[C#]

 

//Executes after the file gets uploaded.

protected void SingleFileUpload_Completed(object sender, EventArgs e)

{

    label.Text = this.UploadBox1.FileName + " has been uploaded successfully";        

}

 

//Executes when the target folder already has a file with the same name.

protected void UploadBox1_FileExists(object sender, EventArgs e)

{

        string sFileName = UploadBox1.FileName;

 

        string sNameWithoutExtension = Path.GetFileNameWithoutExtension( sFileName );

        string sFileExtension = Path.GetExtension( sFileName );       

        for( int i = 1;; i++  )

        {

            string sNewFileName = string.Format( "{0}({1}){2}", sNameWithoutExtension, i, sFileExtension );

            string sNewFullpath = Path.Combine( UploadBox1.UploadFolder, sNewFileName );

            if( !File.Exists( sNewFullpath ) )

            {

                UploadBox1.File.MoveTo( sNewFullpath, false );

                break;

            }

        }

}

 

//Executes before saving the file.

protected bool UploadBox1_BeforeFileSave(object oSender, EventArgs e)

{

bool bSaveInUploadFolder = true;

if( Path.GetExtension( UploadBox1.File.FileName ) == ".zip" )

{

UploadBox1.File.MoveTo( "UserOverrideFolder/" + UploadBox1.File.FileName, true );

//Return False to cancel file saving.

bSaveInUploadFolder = false;

}

return bSaveInUploadFolder;

}

 

[VB.NET]

 

'Executes after the file gets uploaded.

Protected Sub SingleFileUpload_Completed(ByVal sender As Object, ByVal e As EventArgs)

        label.Text = Me.UploadBox1.FileName & "has been uploaded successfully"

End Sub

 

'Executes when the target folder already has a file with the same name.

Protected Sub UploadBox1_FileExists(ByVal sender As Object, ByVal e As EventArgs)

        Dim sFileName As String = UploadBox1.FileName

 

        Dim sNameWithoutExtension As String = Path.GetFileNameWithoutExtension(sFileName)

        Dim sFileExtension As String = Path.GetExtension(sFileName)

        Dim i As Integer = 1

        Do

            Dim sNewFileName As String = String.Format("{0}({1}){2}", sNameWithoutExtension, i, sFileExtension)

            Dim sNewFullpath As String = Path.Combine(UploadBox1.UploadFolder, sNewFileName)

            If (Not File.Exists(sNewFullpath)) Then

                UploadBox1.File.MoveTo(sNewFullpath, False)

                Exit Do

            End If

            i += 1

        Loop

End Sub

 

'Executes before saving the file

Protected Function UploadBox1_BeforeFileSave(ByVal oSender As Object, ByVal e As EventArgs) As Boolean

        Dim bSaveInUploadFolder As Boolean = True

        If Path.GetExtension(UploadBox1.File.FileName) = ".zip" Then

            UploadBox1.File.MoveTo("UserOverrideFolder/" & UploadBox1.File.FileName, True)

            'Return False to cancel file saving.

            bSaveInUploadFolder = False

        End If

        Return bSaveInUploadFolder

End Function
