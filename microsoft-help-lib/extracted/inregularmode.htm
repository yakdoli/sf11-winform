In RegularMode

 

ProgressBar control can be used in regular mode to indicate the users of the progress of the initiated activity.

 

1.   Add the ProgressBar control to the Web Form.

2.   Set the ProgressStyle property to RegularMode.

3.   Here, the process is started on button click invoking a callback of the CallbackPanel. The UpdateProgress method raises the UpdateProgress event.

 

[script]

 

<script language="javascript" type="text/javascript">

    function PerformProgress(button)

    {

        document.getElementById('Button1').disabled=true;

        _sfCallbackPanel1.callback("PeformLongOperation", true);

 

        // Set the value and update the progress before starting the timer.

        // This helps when you are repeating the above callback multiple times.

        _sfProgressBar1.value=0;

        _sfProgressBar1.updateProgress();

        _sfProgressBar1.startProgressTimer();

     }

</script>

<ssw:CallbackPanel ID = "CallbackPanel1" runat = "server" OnCallbackRefresh = "CallbackPanel1_CallbackRefresh">

    <cc1:ProgressBar ID = "ProgressBar1" runat = "server" Width = "232px" BackgroundCss = "Progressbarbgcolor" OnUpdateProgress = "ProgressBar1_UpdateProgress"/>

</ssw:CallbackPanel>

 

4.   The CallbackRefresh event when raised, updates the progress value and the UpdateProgress keeps updating the value of the ProgressPercentage value.

 

[C#]

 

protected void Page_Init(object sender, EventArgs e)

    {

        if (this.IsPostBack == false)

            this.Progress = 0;

    }

 

    // Here we keep track of the progress of the long operation.

    public int Progress

    {

        get

        {

            if (this.Application.Contents["MyProgress"] == null)

                this.Application.Contents["MyProgress"] = 0;

 

            return (int)this.Application.Contents["MyProgress"];

        }

        set

        {

            this.Application.Contents["MyProgress"] = value;

        }

    }

 

    // This is the first callback that gets triggered on button click.

    // This takes nearly 10 seconds due to the Sleep introduced in the code. So, this callback will not return for nearly 10 seconds.

    protected void CallbackPanel1_CallbackRefresh(object sender, Syncfusion.Web.UI.WebControls.CancellableCallbackEventArgs e)

    {

        this.Progress = 0;

        while (true)

        {

            // Simulating a long operation, by making this thread to sleep for 10sec.

            System.Threading.Thread.Sleep(1000);

            this.Progress += 10;

 

            if (this.Progress >= 100)

                break;

        }

 

        this.Progress = 100;

       

        // This script will be processed on the client side after the callback returns the information

        this.CallbackPanel1.AfterCallbackResponseProcessedScript = "document.getElementById('Button1').disabled=false;";

    }

 

    // While the above callback is being processed, the progress bar will continue to fire callbacks to retrieve the progress state.

    // In this sample the progress bar fires these callbacks once every 1sec (controlled by Frequency property whose default value is 1sec).

    // This UpdateProgress event will be raised for each of those callbacks where we provide the progress information to the ProgressBar

    protected void ProgressBar1_UpdateProgress(object sender, EventArgs e)

    {

        this.ProgressBar1.ProgressPercentage = this.Progress;

    }

 

[VB]

 

Private Function Page_Init(ByVal sender As Object, ByVal e As EventArgs) As protectedvoid

    If Me.IsPostBack = False Then

        Me.Progress = 0

    End If

End Function

 

' Here we keep track of the progress of the long operation.

Public Property Progress() As Integer

    Get

        If Me.Application.Contents("MyProgress") Is Nothing Then

            Me.Application.Contents("MyProgress") = 0

        End If

 

        Return CInt(Me.Application.Contents("MyProgress"))

    End Get

    Set(ByVal value As Integer)

        Me.Application.Contents("MyProgress") = Value

    End Set

End Property

 

' This is the first callback that gets triggered on button click.

' This takes nearly 10 seconds due to the Sleep introduced in the code. So, this callback will not return for nearly 10 seconds.

Protected Sub CallbackPanel1_CallbackRefresh(ByVal sender As Object, ByVal e As Syncfusion.Web.UI.WebControls.CancellableCallbackEventArgs)

    Me.Progress = 0

    Do While True

        ' Simulating a long operation, by making this thread to sleep for 10sec.

        System.Threading.Thread.Sleep(1000)

        Me.Progress += 10

 

        If Me.Progress >= 100 Then

            Exit Do

        End If

    Loop

 

    Me.Progress = 100

 

    ' This script will be processed on the client side after the callback returns the information

    Me.CallbackPanel1.AfterCallbackResponseProcessedScript = "document.getElementById('Button1').disabled=false;"

End Sub

 

' While the above callback is being processed, the progress bar will continue to fire callbacks to retrieve the progress state.

' In this sample the progress bar fires these callbacks once every 1sec (controlled by Frequency property whose default value is 1sec).

' This UpdateProgress event will be raised for each of those callbacks where we provide the progress information to the ProgressBar

Protected Sub ProgressBar1_UpdateProgress(ByVal sender As Object, ByVal e As EventArgs)

    Me.ProgressBar1.ProgressPercentage = Me.Progress

End Sub

 

5.   Build and run the project. Click the button to start the progress and it will be disabled. When the process is complete the button will be enabled.
