Through Code

To create Button Control Through Code:

1.   Create a new ASP.NET Web application.

2.   In the .cs file include the following directories.

 

[C#]

using  Syncfusion.Web.UI.WebControls.Shared;

using  Syncfusion.Web.UI.WebControls.Tools;

using  Syncfusion.Web.UI;

 

[VB]

Imports  Syncfusion.Web.UI.WebControls.Shared

Imports  Syncfusion.Web.UI.WebControls.Tools

Imports  Syncfusion.Web.UI

 

3.   In code view, add the following snippets.

 

[C#]    

 

private void Page_Init(object sender, EventArgs e)

{

   BuildRibbon();

}

 

private void BuildRibbon()

{

        Ribbon Ribbon1 = new Ribbon();

        Ribbon1.ApplicationMenuType = Syncfusion.Web.UI.WebControls.Tools.RibbonControl.AppMenuType.ApplicationMenu;

        Ribbon1.Width = 600;

        Ribbon1.ApplicationMenuText = "File";

        Ribbon1.ApplicationMenuColor = "#F9A403";

 

        ApplicationItem AppItem = new ApplicationItem();

       

        ApplicationMenuItem menu1 = new ApplicationMenuItem();

        menu1.Text = "Save";

    

        RibbonTabContext RibbonContext = new RibbonTabContext();

 

        RibbonTab RibbonTab1 = new RibbonTab();       

        RibbonTab1.Text = "Home";

 

        RibbonGroup RibbonGroup1 = new RibbonGroup();

        RibbonGroup1.Text = "Clipboard";

 

        RibbonGroupDivison RibbonDivison1 = new RibbonGroupDivison();

        RibbonGroupDivison RibbonDivison2 = new RibbonGroupDivison();

 

        ButtonAdv Button1 = new ButtonAdv();

        Button1.Text = "Paste";

        Button1.DisplayType = DisplayType.ImageAboveText;

        Button1.ImageUrl = "Img/paste.png";

 

        ButtonAdv Button2 = new ButtonAdv();

        Button2.Text = "Close"

        Button1.DisplayType = DisplayType.ImageAboveText

        Button2.ImageUrl = " Img/Close.png ";

 

       

        AppItem.MenuItem.Item.Add(menu1);

        RibbonDivison1.Content.Add(Button1);

        RibbonDivison1.Content.Add(Button2);

        RibbonDivison2.Content.Add(Button2);

        RibbonGroup1.Division.Add(RibbonDivison1);

        RibbonGroup1.Division.Add(RibbonDivison2);

        RibbonTab1.Groups.Add(RibbonGroup1);

        RibbonContext.Tabs.Add(RibbonTab1);

        Ribbon1.TabContext.Add(RibbonContext);

        form1.Controls.Add(Ribbon1);

}

 

 

 

[VB]    

 

Private ButtonAdvance As New ButtonAdv

 

Private Sub Page_Load(ByVal sender As Object, ByVal e As System.EventArgs) Handles MyBase.Load

        'Put user code to initialize the page here

        BuildButton()

Public Sub BuildButton()

        Dim Ribbon1 As New Ribbon()

        Ribbon1.ApplicationMenuType = Syncfusion.Web.UI.WebControls.Tools.RibbonControl.AppMenuType.ApplicationMenu

        Ribbon1.Width = 600

        Ribbon1.ApplicationMenuText = "File"

        Ribbon1.ApplicationMenuColor = "#F9A403"

 

        Dim AppItem As New ApplicationItem()

 

        Dim menu1 As New ApplicationMenuItem()

        menu1.Text = "Save"

 

        Dim RibbonContext As New RibbonTabContext()

 

        Dim RibbonTab1 As New RibbonTab()

        RibbonTab1.Text = "Home"

 

        Dim RibbonGroup1 As New RibbonGroup()

        RibbonGroup1.Text = "Clipboard"

 

        Dim RibbonDivison1 As New RibbonGroupDivison()

        Dim RibbonDivison2 As New RibbonGroupDivison()

 

        Dim Button1 As New ButtonAdv()

        Button1.Text = "Paste"

        Button1.DisplayType = DisplayType.ImageAboveText

        Button1.ImageUrl = "Img/paste.png"

 

        Dim Button2 As New ButtonAdv()

        Button2.Text = "Close"

                Button1.DisplayType = DisplayType.ImageAboveText

        Button2.ImageUrl = "Img/Close.png"

 

        AppItem.MenuItem.Item.Add(menu1)

        RibbonDivison1.Content.Add(Button1)

        RibbonDivison1.Content.Add(Button2)

        RibbonDivison2.Content.Add(Button2)

        RibbonGroup1.Division.Add(RibbonDivison1)

        RibbonGroup1.Division.Add(RibbonDivison2)

        RibbonTab1.Groups.Add(RibbonGroup1)

        RibbonContext.Tabs.Add(RibbonTab1)

        Ribbon1.TabContext.Add(RibbonContext)

        form1.Controls.Add(Ribbon1)

End Sub
