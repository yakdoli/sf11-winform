Binding to an Array List

 

This section deals with setting an ArrayList as data source to the GridGroupingControl.

 

[C#]

 

protected void Page_Load(object sender, EventArgs e)

{

if (!IsPostBack)

{

ArrayList array = new ArrayList();

array.Add(new MyClass(001, "John David", "UK"));

array.Add(new MyClass(002, "Tom", "USA"));

array.Add(new MyClass(003, "Bretney", "USA"));

array.Add(new MyClass(004, "Jessy", "UK"));

array.Add(new MyClass(005, "Bruch", "USA"));

array.Add(new MyClass(006, "Johny", "USA"));

this.GridGroupingControl1.DataSource = array;

this.GridGroupingControl1.AutoFormat = "Office 2003 Silver";

this.GridGroupingControl1.DataSourceCachingMode = DataSourceCachingMode.ViewState;

}

}

 

// Serializable

public class MyClass

{

    int id;

    string name;

    string city;

    public MyClass(int id, string name, string city)

    {

        this.ID = id;

        this.Name = name;

        this.City = city;

    }

 

    public int ID

    {

        get { return id; }

        set { id = value; }

    }

    public string Name

    {

        get { return name; }

        set { name = value; }

    }

    public string City

    {

        get { return city; }

        set { city = value; }

    }

}

 

[VB]

 

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

If Not IsPostBack Then

Dim array As New ArrayList()

array.Add(New [MyClass](1, "John David", "UK"))

array.Add(New [MyClass](2, "Tom", "USA"))

array.Add(New [MyClass](3, "Bretney", "USA"))

array.Add(New [MyClass](4, "Jessy", "UK"))

array.Add(New [MyClass](5, "Bruch", "USA"))

array.Add(New [MyClass](6, "Johny", "USA"))

Me.GridGroupingControl1.DataSource = array

Me.GridGroupingControl1.AutoFormat = "Office 2003 Silver"

Me.GridGroupingControl1.DataSourceCachingMode = DataSourceCachingMode.ViewState

End If

End Sub
