Integrating Multipage with Tabstrip

 

The Multipage control can be flexibly integrated with TabStrip. This can be used to create user friendly tabbed pages.

 

1.   Open a new ASP.NET Web application.

122.   Drag the MultiPage and Tabstrip controls onto the Web Form.

123.   In the HTML view, add the following coding. To integrate the controls, MultiPageId property of the Tabstrip should be set to the id of the Multipage as shown below.

 

<cc1:TabStrip id = "TabStrip1" runat = "server" ImageBaseURL = "images" CustomCSS = "css/Tab_StdDesign.css" MultiPageID = "MultiPage1" ControlRootPanelCSSClass = " " DynamicPanelCSSClass = " " ControlRootTableCSSClass = " " StaticPanelCSSClass = " ">

 

<LOOKS>

  <cc1:TabStripItemLook id="TabLook">

    <STATEDATADEFAULT RightImageContainerCSSClass = "TabLeftImageCont" TextContainerCSSClass = "TabTextCont" RightImageCSSClass = "TabLeftImageCell" LeftImageContainerCSSClass = "TabLeftImageCont" LeftImageCellCSSClass = "TabLeftImageCont" TextCellCSSClass = "TabTextCell" RightImageCellCSSClass = "TabLeftImageCont" ItemCSSClass = "DefaultTab" LeftImageCSSClass = "TabLeftImageCell" RightImageURL = "tabrighticon.gif" LeftImageURL = "tablefticon.gif"></STATEDATADEFAULT>

 

    <STATEDATAHOVER RightImageContainerCSSClass = "TabLeftImageCont" TextContainerCSSClass = "TabHoverTextCont" RightImageCSSClass = "TabLeftImageCell" LeftImageContainerCSSClass = "TabLeftImageCont" LeftImageCellCSSClass = "TabLeftImageCont" TextCellCSSClass = "TabHoverTextCell" RightImageCellCSSClass = "TabLeftImageCont" ItemCSSClass = "TabHover" LeftImageCSSClass = "TabLeftImageCell" RightImageURL = "hovertabrighticon.gif" LeftImageURL = "hovertablefticon.gif"></STATEDATAHOVER>

 

    <STATEDATAPUSHED RightImageContainerCSSClass = "TabLeftImageCont" TextContainerCSSClass = "TabActiveTextCont" RightImageCSSClass = "TabLeftImageCell" LeftImageContainerCSSClass = "TabLeftImageCont" LeftImageCellCSSClass = "TabLeftImageCont" TextCellCSSClass = "TabActiveTextCell" RightImageCellCSSClass = "TabLeftImageCont" ItemCSSClass = "TabActive" LeftImageCSSClass = "TabLeftImageCell" RightImageURL = "selectedtabrighticon.gif" LeftImageURL = "selectedtablefticon.gif"></STATEDATAPUSHED>

  </cc1:TabStripItemLook>

</LOOKS>

 

<ITEMS>

  <cc1:TabStripItem id="Item1" Look="TabLook" Text="General"></cc1:TabStripItem>

  <cc1:TabStripItem id="Item2" Look="TabLook" Text="Security"></cc1:TabStripItem>

  <cc1:TabStripItem id="Item3" Look="TabLook" Text="Privacy"></cc1:TabStripItem>

  <cc1:TabStripItem id="Item4" Look="TabLook" Text="Content"></cc1:TabStripItem>

  <cc1:TabStripItem id="Item6" Look="TabLook" Text="Programs"></cc1:TabStripItem>

  <cc1:TabStripItem id="Item7" Look="TabLook" Text="Advanced"></cc1:TabStripItem>

</ITEMS>

</cc1:TabStrip>

                

<cc1:MultiPage id="MultiPage1" runat="server" >

  <cc1:PageView id="PageView1">

    <IMG id="general1" src="images/General.gif" runat="server">

  </cc1:PageView>

 

  <cc1:PageView id="PageView2">

    <IMG id="security1" src="images/Security.gif" runat="server">

  </cc1:PageView>

 

  <cc1:PageView id="PageView3">

    <IMG id="privacy1" src="images/Privacy.gif" runat="server">

  </cc1:PageView>

 

  <cc1:PageView id="PageView4">

    <IMG id="Image4" src="images/Content.gif" runat="server">

  </cc1:PageView>

 

  <cc1:PageView id="PageView5">

    <IMG id="Programs1" src="images/Programs.gif" runat="server">

  </cc1:PageView>

 

  <cc1:PageView id="PageView6">

    <IMG id="advanced1" src="images/Advanced.gif" runat="server">

  </cc1:PageView>

</cc1:MultiPage>
