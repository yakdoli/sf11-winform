How to obtain the selected node's text?

 

The selected node's text can be obtained by triggering the NodeSelected event for which the AutoPostBackOnNodeSelect must be set to True.

 

[ASPX]

 

function nodeSelected(node)

{

    document.getElementById('Label1').innerHTML=node.Text;

}

 

<cc1:treeview id = "TreeView1" runat = "server" ClientSideOnNodeSelect= "nodeSelected(this)" >

    <Items>

        .....

    </Items>

</cc1:treeview>

 

<asp:Label ID="Label1" runat="server" Text="Label"></asp:Label>

 

[C#]

 

protected void TreeView1_NodeSelected(object sender, Syncfusion.Web.UI.WebControls.Tools.TreeViewNodeEventArgs e)

{

    Label1.Text = e.Node.Text;

}

 

[VB.NET]

 

Protected Sub TreeView1_NodeSelected(ByVal sender As Object, ByVal e As Syncfusion.Web.UI.WebControls.Tools.TreeViewNodeEventArgs)

       Label1.Text = e.Node.Text

End Sub
