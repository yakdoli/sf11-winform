Using ChartModel

The steps to show a Legend in chart through ChartModel are as follows:

1.   In Controller, create an instance of MVCChartModel.

·      Set the ShowTooltips property of chart model to True  and  set the ChartModel.Calcregions property to True.

·      Set the PointsTooltipTemplateID property of chart model for custom string format.

 

·      Set the ChartSeries, ChartArea, and ChartModel properties.

·      Return view to the corresponding View page after setting the ChartModel to the ViewData.

 

[C#]

public ActionResult SimpleChart()

        {

             ExcelEngine engine = newExcelEngine();

            IApplication app = engine.Excel;

            IWorkbook workBook = app.Workbooks.Open(HttpContext.Request.PhysicalApplicationPath + @"\App_data\ChartData.xlsx");

            IWorksheet sheet = workBook.Worksheets[1];

            DataTable dt = sheet.ExportDataTable(sheet.UsedRange,  ExcelExportDataTableOptions.DetectColumnTypes);

            List<Employee> data = newList<Employee>();

 

            for (int i = 1; i < 7; i++)

            {

                Employee emp = newEmployee();

                emp.EmployeeID = Convert.ToInt32(dt.Rows[i][0]);

                emp.FirstName = Convert.ToString(dt.Rows[i][1]);

                emp.LastName = Convert.ToString(dt.Rows[i][2]);

                emp.Title = Convert.ToString(dt.Rows[i][3]);

                emp.Country = Convert.ToString(dt.Rows[i][4]);

                emp.Rating = Convert.ToInt32(dt.Rows[i][5]);

                data.Add(emp);

            }

 

            ChartModel model = newChartModel();

 

            #region Chart Model Properties

 

            model.Text = "List Chart";

            model.Font = newChartFont("Segoe UI", "14px", ChartFontStyle.Normal);

            model.Size = newSize(1000, 500);

 

            #endregion

 

            #region Series

 

            Series areaSeries = newSeries("Area Series");

            Random ran = newRandom();

            areaSeries.Type = SeriesType.Area;

            areaSeries.BindTo = newChartDataBind() { DataSource = data, XName = "EmployeeID", YNames = newstring[] { "Rating" } };

 

 

            model.Series.Add(areaSeries);

         

         //Enabling the tooltip support

            model.ShowToolTip = true;

            model.CalcRegion = true;

            areaSeries.PointsTooltipTemplateID = "Tooltip";

 

            model.ClientSideEvents.OnLoad ="OnLoad";

            model.ClientSideEvents.OnPointRegionMouseMove = "OnMouseMove";

 

            ViewData["ChartModel"] = model;

 

            return View();

 

        }

[JavaScript]

 

 <scripttype="text/javascript">

        var chartModel = null;

        $(document).ready(function () {

            if (!$.browser.msie)

                $("#Tooltip").find("tr:first").remove();

            $(document).bind("mousemove", function (e) {

                if ($(e.target).hasClass("tooltipDiv") || $(e.target).parents(".tooltipDiv").length > 0) {

                    $(".tooltipDiv").remove();

                }

            });

        });

        function OnLoad(sender, args) {

            chartModel = sender.model;

        }

        function OnMouseMove(sender, args) {

            var series = chartModel.Series[args.Data.Region.SeriesIndex];

            var point = series.Points[args.Data.Region.Region.PointIndex];

            var data = $.Enumerable.From(series.BindTo.DataSource).Where(function (p) {

                return point.X == p.EmployeeID;

            }).ToArray();

 

            $(".FirstName").html(data[0].FirstName);

            $(".LastName").html(data[0].LastName);

            $(".Title").html(data[0].Title);

            $(".Country").html(data[0].Country);

            $(".Rating").html(data[0].Rating);

 

        }

    </script>
