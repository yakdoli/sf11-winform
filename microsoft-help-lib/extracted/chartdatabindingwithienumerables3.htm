Chart Data Binding with IEnumerables

You can bind the chart to IEnumerable data sources using the following code.

 

[C#]     

        ExcelEngine engine = new ExcelEngine();

        IApplication app = engine.Excel;

        IWorkbook workBook = app.Workbooks.Open(Request.PhysicalApplicationPath + @"App_data\ChartData.xlsx");

        IWorksheet sheet = workBook.Worksheets[0];

        DataTable dt = sheet.ExportDataTable(sheet.UsedRange, ExcelExportDataTableOptions.DetectColumnTypes);

        List<Chartdata> data = new List<Chartdata>();

 

        for (int i = 1; i < 100; i++)

            data.Add(new Chartdata(Convert.ToDateTime(dt.Rows[i][0]).ToOADate(), Convert.ToDouble(dt.Rows[i][1])));

              

        #region Chart  this.ChartAdv1 Properties

      

         this.ChartAdv1.Text = "List Chart";

        

         this.ChartAdv1.Size = new Size(1000, 500);

 

        #endregion

 

        #region Series

 

        Series areaSeries = new Series("Area Series");

        Random ran = new Random();

        areaSeries.Type = SeriesType.Area;

        areaSeries.BindTo = new ChartAdvDataBind() { DataSource = data, XName = "Date", YNames = new string[] { "YValue" } };

 

 

         this.ChartAdv1.Series.Add(areaSeries);

 

        areaSeries.Style.Border.Width = 5;

        areaSeries.Style.Border.Color = Color.FromArgb(51, 119, 181);

        areaSeries.Style.Interior = new GradientInfo(new ColorInfo(Color.FromArgb(241, 248, 254)));

 

 

        #endregion

 

        #region Axes Customization

         this.ChartAdv1.Axes["PrimaryX"].ValueType = ChartAxisValueType.DateTime;

         this.ChartAdv1.Axes["PrimaryX"].RangeType = RangeType.Set;

         this.ChartAdv1.Axes["PrimaryX"].Range = new MinMaxInfo() { Start = 40126, End = 40812, Interval = 115 };

         this.ChartAdv1.Axes["PrimaryX"].DateFormat = "mm/dd/ yyyy";

 

         this.ChartAdv1.Axes["PrimaryY"].RangeType = RangeType.Set;

         this.ChartAdv1.Axes["PrimaryY"].Range = new MinMaxInfo() { Start = 0, End = 90, Interval = 10 };

        #endregion

 

public class Chartdata

    {

        public Chartdata(double date, double yValue)

        {

            this.Date = date;

            this.YValue = yValue;

        }

        public double Date

        {

            get;

            set;

        }

        public double YValue

        {

            get;

            set;

        }

 

    }

 

 

 

[VB]     

        Dim engine As ExcelEngine = New ExcelEngine()

        Dim app As IApplication = engine.Excel

        Dim workBook As IWorkbook = app.Workbooks.Open(Request.PhysicalApplicationPath & "App_data\ChartData.xlsx")

        Dim sheet As IWorksheet = workBook.Worksheets(0)

        Dim dt As DataTable = sheet.ExportDataTable(sheet.UsedRange, ExcelExportDataTableOptions.DetectColumnTypes)

        Dim data As List(Of Chartdata) = New List(Of Chartdata)()

 

        For i As Integer = 1 To 99

            data.Add(New Chartdata(Convert.ToDateTime(dt.Rows(i)(0)).ToOADate(), Convert.ToDouble(dt.Rows(i)(1))))

        Next i

 

 

        Me.ChartAdv1.Text = "List Chart"

 

        Me.ChartAdv1.Size = New Size(1000, 500)

        

 

        Dim areaSeries As Series = New Series("Area Series")

        Dim ran As Random = New Random()

        areaSeries.Type = SeriesType.Area

      

        areaSeries.BindTo = New ChartAdvDataBind() With {.DataSource = data, .XName = "VDate", .YNames = New String() {"YValue"}}

 

 

        Me.ChartAdv1.Series.Add(areaSeries)

 

        areaSeries.Style.Border.Width = 5

        areaSeries.Style.Border.Color = Color.FromArgb(51, 119, 181)

        areaSeries.Style.Interior = New GradientInfo(New ColorInfo(Color.FromArgb(241, 248, 254)))

 

 

     

        Me.ChartAdv1.Axes("PrimaryX").ValueType = ChartAxisValueType.DateTime

        Me.ChartAdv1.Axes("PrimaryX").RangeType = RangeType.Set

      

        Me.ChartAdv1.Axes("PrimaryX").Range = New MinMaxInfo() With {.Start = 40126, .End = 40812, .Interval = 115}

        Me.ChartAdv1.Axes("PrimaryX").DateFormat = "mm/dd/ yyyy"

 

        Me.ChartAdv1.Axes("PrimaryY").RangeType = RangeType.Set

             

        Me.ChartAdv1.Axes("PrimaryY").Range = New MinMaxInfo() With {.Start = 0, .End = 90, .Interval = 10}

 

 

Public Class Chartdata

        Public Sub New(ByVal [date] As Double, ByVal yValue As Double)

 

            Me.VDate = [date]

            Me.YValue = yValue

        End Sub

        Private privateDate As Double

        Public Property VDate() As Double

            Get

                Return privateDate

            End Get

            Set(ByVal value As Double)

                privateDate = value

            End Set

        End Property

        Private privateYValue As Double

        Public Property YValue() As Double

            Get

                Return privateYValue

            End Get

            Set(ByVal value As Double)

                privateYValue = value

            End Set

        End Property

 

    End Class
