ObservableCollection Data Binding

 

Chart Series that is bound using Observable Collection, automatically gets updated when any changes are made in the collections. Chart can support binding data of either ObservableCollection or INotifyCollectionChanged collections.

 

The following lines of code can be used to bind an ObservableCollection of instances to chart series.

 

[XAML]

 

<UserControl.Resources>

        <util:Sports x:Name="SportsData"/>

    </UserControl.Resources>

                <syncfusion:ChartArea>

                    <!--Initialize the primary axis LabelSource as Observable collection object-->

                    <syncfusion:ChartArea.PrimaryAxis>

                        <syncfusion:ChartAxis IsAutoSetRange="True" LabelsSource="{StaticResource SportsData}" ContentPath="SportName" PositionPath="SportID"/>

                    </syncfusion:ChartArea.PrimaryAxis>

 

                    <!--Initialize the ChartSeries DataSource as Observable collection object-->

                    <syncfusion:ChartSeries Type="Column" Name="Series1" Interior="LightSkyBlue" DataSource="{StaticResource SportsData}" BindingPathX="SportID" BindingPathsY="Interest" />

                </syncfusion:ChartArea>

 

[C#]

 

public class Sport

{

      public int SportID { get; set; }

      public string SportName { get; set; }

      public double Interest { get; set; }

}

 

 

public class Sports : ObservableCollection<Sport>

{

      public Sports()

      : base()

      {

            Add(new Sport() { SportID = 101, SportName = "Golf", Interest = 9 });

            Add(new Sport() { SportID = 102, SportName = "Soccer", Interest = 40 });

            Add(new Sport() { SportID = 103, SportName = "Hockey", Interest = 15 });

            Add(new Sport() { SportID = 104, SportName = "Rugby", Interest = 7 });

            Add(new Sport() { SportID = 105, SportName = "Atheletics", Interest = 3 });

            Add(new Sport() { SportID = 106, SportName = "Tennis", Interest = 10 });

            Add(new Sport() { SportID = 107, SportName = "Baseball", Interest = 6 });

            Add(new Sport() { SportID = 108, SportName = "Basketball", Interest = 10 });

      }

}
