Multiple Axes

 

Chart can render multiple axes in the same plot. Different series can be plotted against different axes as shown below.

 

[XAML]

 

<syncfusion:Chart x:Name="MyChart" Height="500" Background="{StaticResource chartbackground}" Margin="15">

 

     <syncfusion:ChartArea BorderBrush="Black" Margin="5">

             <syncfusion:ChartArea.Header>

          <TextBlock Text="Product Review" FontFamily="Arial" FontSize="20" Foreground="Blue" TextAlignment="Center"/>

                    </syncfusion:ChartArea.Header>

 

                    <syncfusion:ChartArea.Legends>

                        <syncfusion:ChartLegend/>

                    </syncfusion:ChartArea.Legends>

 

                    <syncfusion:ChartArea.PrimaryAxis>

                        <syncfusion:ChartAxis Interval="1" IsAutoSetRange="True" ShowGridLines="False">

                            <syncfusion:ChartAxis.Header>

                                <TextBlock Text="Product ID" FontSize="12" Margin="3"/>

                            </syncfusion:ChartAxis.Header>

                        </syncfusion:ChartAxis>

                    </syncfusion:ChartArea.PrimaryAxis>

 

          <syncfusion:ChartArea.SecondaryAxis>

          <syncfusion:ChartAxis Header="Sales" Interval="2" OpposedPosition="False" ShowGridLines="False"/>

                    </syncfusion:ChartArea.SecondaryAxis>

 

          <syncfusion:ChartSeries  Type="Area" Name="Series1" Label="Sales" Interior="{StaticResource Series1}"/>

          <syncfusion:ChartSeries Type="Area" Name="Series2" Label="Profit" Interior="{StaticResource Series2}">

                        <syncfusion:ChartSeries.YAxis>

          <syncfusion:ChartAxis Header="Profit" Interval="2" OpposedPosition="True" ShowGridLines="False"/>

                        </syncfusion:ChartSeries.YAxis>

                    </syncfusion:ChartSeries>

 

                    <syncfusion:ChartSeries Type="Area" Name="Series3" Label="Production" >

                        <syncfusion:ChartSeries.YAxis>

                  <syncfusion:ChartAxis Header="Production" Interval="2" OpposedPosition="True" ShowGridLines="False"/>

                        </syncfusion:ChartSeries.YAxis>

                    </syncfusion:ChartSeries>

                </syncfusion:ChartArea>

            </syncfusion:Chart>
