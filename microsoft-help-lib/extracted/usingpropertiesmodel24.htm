Using Properties Model

The following steps show you how to handle the addition of a symbol palette group through the properties model.

1.   In the controller, create an object for the DiagramPropertiesModel class. Assign this model class to view data. 

 

Controller 

DiagramPropertiesModel model = new DiagramPropertiesModel();

        public ActionResult FlatDiagram()

        {

            SymbolPaletteItem item1 = new SymbolPaletteItem();

            item1.ContentId = "Image_Content";

            item1.Name = "Image";

 

            SymbolPaletteItem item2 = new SymbolPaletteItem();

            item2.ContentId = "Slider_Content";

            item2.Name = "Slider";

 

            SymbolPaletteItem item3 = new SymbolPaletteItem();

            item3.ContentId = "DatePicker_Content";

            item3.Name = "DatePicker";

 

            SymbolPaletteItem item4 = new SymbolPaletteItem();

            item4.ContentId = "TextBox_Content";

            item4.Name = "TextBox";

 

            SymbolPaletteGroup g1 = new SymbolPaletteGroup();

            g1.HeaderName = "Custom Shapes";

            g1.Index = 2;

            g1.Items.Add(item1);

            g1.Items.Add(item2);

            g1.Items.Add(item3);

            g1.Items.Add(item4);

 

            model.SymbolPalette.Groups.Add(g1);

            model.Width = 900;

            model.Height = 500;

            model.DiagramMode = DiagramMode.Canvas;

            ViewData["FlatDiagram"] = model;

            return View();

        }

 

 

 

2.   In the view, invoke the Diagram helper with the view data key as the control ID.

 

View

 

<div id="Image_Content" class="nodecontent" style="width: 110px;height: 100px">

  <img alt="Image_Content" 

src="<%=Url.Content("~/Content/Images/NodeContents/Callahan.png")%>" />

</div>

 

<div id="Slider_Content" class="nodecontent" style="width: 110px;

height: 25px; padding: 2px;">

        <div style="width: 110px; height: 25px">

            <%=Html.Syncfusion().Slider("sliderContent")

        .Minimum(0)

        .Maximum(100)

        .Step(1)

        .Value(10)

        .AutoFormat(Skins.Office2007Blue)

            %>

        </div>

</div>

 

<div id="DatePicker_Content" class="nodecontent"

 style="width: 215px; height: 210px">

        <%{

              Html.Syncfusion().DatePicker("datePickerContent")

                 .AutoFormat(Skins.Office2007Blue)

                 .Inline(true).Render();

          }%>

</div>

 

<div id="TextBox_Content" class="nodecontent">

        <%=Html.TextBox("textboxContent","I am a textbox",new { style="width:120px;height:25px;" })%>

</div>

 

<%{
      Html.Syncfusion().Diagram("FlatDiagram")

          .Render();

  }
%>

 

3.   Build and run the application.
