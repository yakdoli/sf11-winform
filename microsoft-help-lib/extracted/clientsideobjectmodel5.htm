protected void Page_Load(object sender, EventArgs e)

    {

        if(!IsPostBack)

    protected void TreeView1_CallbackRefresh(object sender, Syncfusion.Web.UI.WebControls.Shared.CallbackEventArgs e)

    {

        string sCallbackArgument = e.CallbackArgument;

        int nCommandLength = sCallbackArgument.IndexOf('|');

        string sArgument = sCallbackArgument.Substring( nCommandLength+1 );

        string sCommand = sCallbackArgument.Substring(0,nCommandLength);

 

        switch (sCommand)

        {

 case "addroot":

                TreeViewNode tvnNewRootNode = new TreeViewNode(sArgument);

                TreeView1.Items.Add(tvnNewRootNode);

                TreeView1.SelectedNode = tvnNewRootNode;

                break;

 

            case "addchild":

                TreeViewNode tvnNewNode = new TreeViewNode(sArgument);

 

                if (null != TreeView1.SelectedNode)

                {

                    TreeView1.SelectedNode.Items.Add(tvnNewNode);

                    TreeView1.SelectedNode.Expanded = true;

                }

                else

                {

                    TreeView1.Items.Add(tvnNewNode);

                }

 

                TreeView1.SelectedNode = tvnNewNode;

                break;

 

            case "rename":

                if (null != TreeView1.SelectedNode)

                {

                    TreeView1.SelectedNode.Text = sArgument;

                }

                break;

 

            case "remove":

                if (null != TreeView1.SelectedNode)

                {

                    TreeViewItemCollection tviCollection = (null != TreeView1.SelectedNode.ParentItem) ?

                        TreeView1.SelectedNode.ParentItem.Items : TreeView1.Items;

 

                                  TreeViewNode tvnParentNode = TreeView1.SelectedNode.ParentItem;

                    tviCollection.Remove(TreeView1.SelectedNode);

 

                                  if( null != tvnParentNode )

                                  {

                                         TreeView1.SelectedNode = tvnParentNode;

                                  }

                                  else if( 0 < TreeView1.Items.Count )

                                  {

                                         TreeView1.SelectedNode = TreeView1.Items[0];

                                  }

                }

                break;

        }

 

              TextField.Value = ( null != TreeView1.SelectedNode ) ? TreeView1.SelectedNode.Text : "";

    }

function GetNodesCount()

{

    var nCount = GetChildNodesCountRecursive( null );

    return nCount;

}

 

function GetChildNodesCountRecursive( oNode )

{

    var nCount = 0;

    var arrChildNodes = ( null == oNode ) ? _sfTreeView1.GetRootNodes() : oNode.Nodes;

    if( null != arrChildNodes )

    {

        for( var i = 0; i < arrChildNodes.length; i++ )

        {

            var oChildNode = arrChildNodes[i];

            nCount++;

            nCount += GetChildNodesCountRecursive( oChildNode );

        }

    }

    return nCount

}

 

function ShowAllNodesCount()

{

    var nCount = GetNodesCount();

    alert( "All nodes count: " + nCount);

function CheckNodesRecursive( oNode, bCheck )

{

    var arrChildNodes = ( null == oNode ) ? _sfTreeView1.GetRootNodes() : oNode.Nodes;

    if( null != arrChildNodes )

    {

        for( var i = 0; i < arrChildNodes.length; i++ )

        {

            var oChildNode = arrChildNodes[i];

            if( bCheck )

            {

                oChildNode.Check();

            }

            else

            {

                oChildNode.UnCheck();

            }

            CheckNodesRecursive( oChildNode, bCheck );

        }

    }

}

The ClientObjectID can be used to access the control's object model on the client side.

 

ClientObjectID can be effectively used to refer the control's objects when used with MasterPages and UserControls. function CheckNodesRecursive( oNode, bCheck )

{

    var arrChildNodes = ( null == oNode ) ? _sfTreeView1.GetRootNodes() : oNode.Nodes;

    if( null != arrChildNodes )

    {

        for( var i = 0; i < arrChildNodes.length; i++ )

        {

            var oChildNode = arrChildNodes[i];

            if( bCheck )

            {

                oChildNode.Check();

            }

            else

            {

                oChildNode.UnCheck();

            }

            CheckNodesRecursive( oChildNode, bCheck );

        }

    }
