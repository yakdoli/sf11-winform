Client-Side Events

 

 

Client side events can be triggered and handled that allows to access and control the application on client side.

The client side events of GenericDropDownCombo are listed below.

 

Client-Side Event

Description

ClientSideOnBeforePopupShow

Specifies the client side function to call before the popup is shown.

ClientSideOnPopupShown

Specifies the client side function to call when the popup is shown.

ClientSideOnTextChanged

Specifies the client side function to call when the text is changed.

<script type="text/javascript">

   

  function SetDropDownText()

  {       

    var text="";

 

    if(document.getElementById("EnterpriseBin").checked==true)

    {

      text+=document.getElementById("EnterpriseBin").value;

    }

    if(document.getElementById("EnterpriseSource").checked==true)

    {

      text+=document.getElementById("EnterpriseSource").value;

    }

    if(document.getElementById("ASPBin").checked==true)

    {

      text+=document.getElementById("ASPBin").value;

    }

    if(document.getElementById("ASPSource").checked==true)

    {

      text+=document.getElementById("ASPSource").value;

    }

    if(document.getElementById("WindowsBin").checked==true)

    {

      text+=document.getElementById("WindowsBin").value;

    }

    if(document.getElementById("WindowsSource").checked==true)

    {

      text+=document.getElementById("WindowsSource").value;

    }

    if(document.getElementById("BackOfficeBin").checked==true)

    {

      text+=document.getElementById("BackOfficeBin").value;

    }

    if(document.getElementById("BackOfficeSource").checked==true)

    {

      text+=document.getElementById("BackOfficeSource").value;

    }

    Pricelistcombo.SetText(text);

  }

 

  function OnBeforePopup()

  {

    document.getElementById("ClientEventbox").innerHTML="";

    document.getElementById("ClientEventbox").innerHTML+="ClientSideOnBeforePopupShow event is fired <br><br>";

  }

 

  function OnPopup()

  {

    document.getElementById("ClientEventbox").innerHTML+="ClientSideOnPopupShown event is fired <br><br>";

  }

  function OnTextChangePopup()

  {

    document.getElementById("ClientEventbox").innerHTML+="ClientSideOnTextChanged event is fired <br><br>";

  }   

</script>
