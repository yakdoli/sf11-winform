Using Builder

The steps to implement tooltip support for the Chart series through Builder are as follows:

1.   In Controller, return view to the corresponding View page.

 

[C#]

   public ActionResult SimpleChart()

   {

       return View();

   }

 

 

2.   In the View page, invoke the ChartBuilder by using the control ID as the first argument.

3.   Set the ShowTooltips property of chart model to True  and  set the ChartModel.Calcregions property to True.

 

[ASPX]

<% = Html.Syncfusion.Chart("Chart")

              .Series(series =>

              {

                  series.Add().Points(p =>

                      {

                          p.Add(1910, 93);

                          p.Add(1920, 106);

                          . . .

                      })

                      .Name("Rating").Type(SeriesType.Spline).Symbol(symbol =>

                      {

                          symbol.Visible(true).Shape(SymbolShape.Star);

                      });

               

 

              }).Axes(axes =>

              {

                  axes.PrimaryX(xaxis =>

                  {

                      xaxis.Title("Year").MajorGridLines(line => line.Visible(false)).Font(newChartFont("Arial", "15px", ChartFontStyle.Bold));

                  }).PrimaryY(yaxis =>

                  {

                      yaxis.Title("Monthly Rating in %").Font(new Syncfusion.Mvc.ChartAdv.ChartFont("Arial", "15px", ChartFontStyle.Bold));

                  });

              }).Size(newSize(1000, 500)).Text("Employee Details").Font(newChartFont("Arial", "15px", ChartFontStyle.Bold)).Legend(legend => legend.Visible(true).Shape(LegendShape.Triangle)).ElementSpacing(10)

             //Enabling the tooltip support

                 .CalcRegion(true).ShowToolTip(true)

             

    %>
