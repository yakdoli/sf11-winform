Using Builder

The following steps show you how to handle the addition of a custom shape to the diagram page through Builder.

1.   In the view, invoke the Diagram helper with the control ID as the first argument. 

 

View

 

<% Node node1 = new Node()

   {

       Name = "Node1",

       Label = "Node1",

   };

   

   Node node2 = new Node()

   {

       Name = "Node2",   

       Shape = Shapes.CustomShape

   };

   Node node3 = new Node()

   {

       Name = "Node3",

       Shape = Shapes.CustomShape

   };

   LineConnector line1 = new LineConnector()

   {

       Name = "line1",

       HeadNode = node1,

       TailNode = node2,

       HeadDecoratorShape = DecoratorShape.None,

       TailDecoratorShape = DecoratorShape.None,

       ConnectorType = ConnectorType.Orthogonal,

       LineColor = "black",

       LineWidth = 2

   };

   LineConnector line2 = new LineConnector()

   {

       Name = "line2",

       HeadNode = node1,

       TailNode = node3,

       HeadDecoratorShape = DecoratorShape.None,

       TailDecoratorShape = DecoratorShape.None,

       ConnectorType = ConnectorType.Orthogonal,

       LineColor = "black",

       LineWidth = 2

   };

<%{

      Html.Syncfusion().Diagram("FlatDiagram")

          .Nodes(nodes => nodes.Add(node1).Add(node2).Add(node3))

          .Connectors(conn => conn.Add(line1).Add(line2))

          .LayoutType(LayoutType.DirectedTreeLayout)

          .Width(900)

          .Height(500)

          .DiagramMode(DiagramMode.SVG)

          .ClientSideOnCustomShapeDrawing("OnCustomShapes")

          .Render();

  }

%>

 

 

Note: If you want to create the diagram in the Canvas mode, change the DiagramMode to Canvas. By default the diagram is rendered in the SVG mode.

 

[JavaScript] 

<script type="text/javascript">

       function OnCustomShapes (sender,args) {

          // args:

          // _Context    - Canvas context of the custom shape node

          // _Width      - node width of the custom shape

          // _Height     - node Height of the custom shape

          // _NodeName   – name of the custom shape node

          // _LineWidth; - linewidth of the custom shape node

 

        }

</script> 

 

 

2.   Build and run the application
