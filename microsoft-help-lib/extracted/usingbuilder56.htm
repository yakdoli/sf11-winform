Using Builder

 

The steps to customize a Symbol in chart through Builder are as follows:

1.   In Controller, return view to the corresponding View page.

 

[C#]

   public ActionResult SimpleChart()

   {

       return View();

   }

 

 

2.   In the View page, invoke the ChartBuilder by using the control ID as the first argument.

3.   Set the Visible property of Symbol to True and set the style.

 

[ASPX]

 

    <% ass.Start = newDateTime(2010, 1, 1).ToOADate(); %>

    <%ass.End = newDateTime(2010, 12, 1).ToOADate(); %>

    <%ass.Interval = 31; %>

    <%string skyImage = UrlHelper.GenerateContentUrl("~/Content/Images/sky.jpg", Html.ViewContext.HttpContext).ToString(); %>

    <%string coolImage = UrlHelper.GenerateContentUrl("~/Content/Images/cool.png", Html.ViewContext.HttpContext).ToString(); %>

    <%string hotImage = UrlHelper.GenerateContentUrl("~/Content/Images/hot.png", Html.ViewContext.HttpContext).ToString(); %>

    <%= Html.MobSyncfusion().Chart("Chart")

              .Series(series =>

              {

                  series.Add().Name("London").Type(SeriesType.Line).Points(p =>

                  {

                      p.Add(newDateTime(2010,1,1), 7);

                      p.Add(newDateTime(2010, 2, 1), 6.9);

                      p.Add(newDateTime(2010, 3, 1), 10);

                      p.Add(newDateTime(2010, 4, 1), 14.5);

                      p.Add(newDateTime(2010, 5, 1), 18.4);

                      p.Add(newDateTime(2010, 6, 1), 21.5);

                      p.Add(newDateTime(2010, 7, 1), 25.2);

                      p.Add(newDateTime(2010, 8, 1), 26.5).Symbol(symbol => {

                          symbol.ImageUrl(hotImage).Shape(SymbolShape.Image).Visible(true).Size(newSize(62,62));

                        });

                      p.Add(newDateTime(2010, 9, 1), 23.6);

                      p.Add(newDateTime(2010, 10, 1), 17);

                      p.Add(newDateTime(2010, 11, 1), 13.5);

                      p.Add(newDateTime(2010, 12, 1), 12);

 

                  }).Style(style =>

                  {

                      style.Border(border => border.Width(3)).Opacity(0.8f);

                  }).Symbol(symbol =>

                  {

                      symbol.Shape(SymbolShape.Circle).Style(style => style.Interior(newGradientInfo(newColorInfo(Color.White)))).Visible(true);

                  });

 

                  series.Add().Name("Tokyo").Type(SeriesType.Line).Points(p =>

                  {

                      p.Add(newDateTime(2010, 1, 1), 4).Symbol(symbol =>

                      {

                          symbol.ImageUrl(coolImage).Shape(SymbolShape.Image).Visible(true).Size(newSize(62,62));

                      });

                      p.Add(newDateTime(2010, 2, 1), 4.2);

                      p.Add(newDateTime(2010, 3, 1), 5.7);

                      . . .

                  }).Style(style =>

                  {

                      style.Border(border => border.Width(3)).Opacity(0.8f);

                  }).Symbol( symbol =>

                      {

                          symbol.Shape(SymbolShape.Circle).Style(style => style.Interior(newGradientInfo(newColorInfo(Color.White)))).Visible(true);

                      });

              

              })

    %>
