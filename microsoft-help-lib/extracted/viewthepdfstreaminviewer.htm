View the PDF stream in viewer?

PDF files as stream can be viewed in Essential Pdf Viewer using the property Stream. Use the following code snippets to load the PDF streams in viewer:

 

 

[Controller]

public class HomeController : Controller

    {      

 

        public ActionResult Index()

        {

            var model= new PdfViewerModel();

          FileStream stream = new FileStream("Template.pdf", FileMode.Open);

         

            model.Stream =stream;

            ViewData["myModel"]=model;

            return View();

        }

 

        [AcceptVerbs(HttpVerbs.Post)]

        public ActionResult Index(PdfViewerParams args)

        {

            var model= new PdfViewerModel();          

            model.Stream =stream;           

            return model.PdfViewerHtmlActionResult(args);

        }

    }

 

 

 

Add the following code in View:

[ASPX]

 

<%@ Page Language="C#" MasterPageFile="~/Views/Shared/Site.Master" Inherits="System.Web.Mvc.ViewPage" %>

 

<asp:Content ID="Content1" ContentPlaceHolderID="TitleContent" runat="server">

   PdfViewer Model

</asp:Content>

 

<asp:Content ID="Content2" ContentPlaceHolderID="MainContent" runat="server">

 

<%--Rendering the PdfViewer Control--%>

   

    <%= Html.Syncfusion().PdfViewer("PdfViewerModel",(PdfViewerModel)ViewData["myModel"]) %>

 

</asp:Content>
