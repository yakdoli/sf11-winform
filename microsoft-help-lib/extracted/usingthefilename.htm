Using the file name

The following code snippets illustrate the implementation of saving the diagram page using the file name:

Controller

 

DiagramPropertiesModel serializeModel = new DiagramPropertiesModel();

 

public ActionResult Serialization()

{

    this.GetNodesConnectors(serializeModel);

    serializeModel.Height = 650;

    serializeModel.Width = 800;

    serializeModel.DiagramMode = DiagramMode.SVG;

    serializeModel.IsSymbolPaletteEnabled = false;

    serializeModel.BoundaryConstraintsEnabled = true;

    serializeModel.SaveXmlMapper = "SaveDiagramAsXml";

    ViewData["Serialization"] = serializeModel;

    return View();

}

 

/// <summary>

/// Gets the nodes connectors.

/// </summary>

/// <param name="serializeModel">The serialize model.</param>

private void GetNodesConnectors(DiagramPropertiesModel serializeModel)

{

    Node tv = AddNodeSerialize("tv", "", 350, 50, 103.00, 103.00, "", 1);

    Node seat = AddNodeSerialize("seat", "", 100, 150, 103.00, 103.00, "", 1);

    Node speaker = AddNodeSerialize("speaker", "", 550.00, 150, 103.00, 103.00, "", 1);

    Node home = AddNodeSerialize("home", "", 350.00, 250.00, 103.00, 110.00, "", 1);

    Node garden = AddNodeSerialize("garden", "", 100.00, 350.00, 103.00, 103.00, "", 1);

    Node paint = AddNodeSerialize("paint", "", 550, 350.00, 103.00, 103.00, "", 1);

    Node cooking = AddNodeSerialize("cooking", "", 350, 500.00, 103.00, 103.00, "", 1);

    LineConnector line1 = AddConnectorSerialize("line1", tv, home, ConnectorType.Straight);

    LineConnector line2 = AddConnectorSerialize("line2", seat, home, ConnectorType.Orthogonal);

    LineConnector line3 = AddConnectorSerialize("line3", speaker, home, ConnectorType.Orthogonal);

    LineConnector line4 = AddConnectorSerialize("line4", garden, home, ConnectorType.Orthogonal);

    LineConnector line5 = AddConnectorSerialize("line5", paint, home, ConnectorType.Orthogonal);

    LineConnector line6 = AddConnectorSerialize("line6", cooking, home, ConnectorType.Straight);

    serializeModel.Nodes = new NodesCollection()

    {

        home, tv, seat, speaker, garden, paint, cooking

    };

    serializeModel.Connectors = new LinesCollection()

    {

        line1, line2, line3, line4, line5, line6

    };

}

 

public ActionResult SaveDiagramAsXml(string FileName)

{

    return serializeModel.Save(FileName);

}

public Node AddNodeSerialize(string name, string label, double offsetX, double offsetY, double width, double height, string bgColor, double bWidth)

{

    Node node = new Node()

    {

        Name = name,

        Label = "",

        LabelHorizontalAlignment = Horizontal.Center,

        LabelVerticalAlignment = Vertical.Middle,

        OffsetX = offsetX,

        OffsetY = offsetY,

        Shape = Shapes.Default,

        Height = height,

        Width = width,

        BackgroundColor = bgColor != null && bgColor != "" ? bgColor : "#fff",

        BorderColor = bgColor != null && bgColor != "" ? bgColor : "#7a7772",

        BorderWidth = bWidth,

        LabelFontColor = "black",

        LabelFontSize = 12,

        LabelFontFamily = "Times New Roman",

        LabelLetterSpacing = 4,

        ContentId = name + "_Content"

    };

    return node;

}

 

public LineConnector AddConnectorSerialize(string name, Node headNode, Node tailNode, ConnectorType connectorType)

{

    LineConnector lineConnector = new LineConnector()

    {

        Name = name,

        HeadNode = headNode,

        TailNode = tailNode,

        HeadDecoratorShape = DecoratorShape.None,

        TailDecoratorShape = DecoratorShape.None,

        ConnectorType = connectorType,

        LineColor = "#7a7772",

        LineWidth = 2

    };

    return lineConnector;

}
