Load On Demand

 

TreeView nodes can be loaded on demand that enables to load the tree nodes only when required. This helps to reduce the round trip time by loading only the required nodes.

Load on demand feature allows the data to be retrieved only on some user action by considerably reducing the downtime/round trip by avoiding postbacks and improve the performance of the application.

 

 

Figure 187

 

To add nodes on demand, the following sample code behind file illustrates the feature.

 

When a node is clicked, postback is avoided and the nodes are built by calling the BuildTree method, which creates the tree structure by loading the nodes on demand.

 

[C#]

 

protected void Page_Load(object sender, EventArgs e)

{

    if (!IsPostBack)

    {

        tview = TreeView1;

        tview.Items.Add(this.BuildTree());

        TreeViewNode node = new TreeViewNode();

        node.Text = "LoadOnDemand";

        node.ImagePath = "folders.gif";

        node.ExpandMode = TreeViewNodeExpandMode.ServerSideCallback;

        tview.Items.Add(node);

    }

}

 

protected void TreeView1_NodeExpanded(object sender, Syncfusion.Web.UI.WebControls.Tools.TreeViewNodeEventArgs e)

{

    e.Node.Items.Add(this.BuildTree());

    TreeViewNode node = new TreeViewNode();

    node.Text = "LoadOnDemand";

    node.ImagePath = "folders.gif";

    node.ExpandMode = TreeViewNodeExpandMode.ServerSideCallback;

    e.Node.Items.Add(node);

}

 

private TreeViewNode BuildTree()

{

 

    TreeViewNode node = new TreeViewNode();

    node.Text = "Static Tree";

    node.ImagePath = "folders.gif";

    node.Expanded = true;

   

 

    TreeViewNode childNode = new TreeViewNode();

    childNode.Text = "Calendar";

    childNode.ImagePath = "calendar.gif";

 

    node.Items.Add(childNode);

 

    childNode = new TreeViewNode();

    childNode.Text = "Notes";

    childNode.ImagePath = "notes.gif";

 

    node.Items.Add(childNode);

    return node;

}

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

       If (Not IsPostBack) Then

              tview = TreeView1

              tview.Items.Add(Me.BuildTree())

              Dim node As TreeViewNode = New TreeViewNode()

              node.Text = "LoadOnDemand"

              node.ImagePath = "folders.gif"

              node.ExpandMode = TreeViewNodeExpandMode.ServerSideCallback

              tview.Items.Add(node)

       End If

End Sub

Private Function BuildTree() As TreeViewNode

 

       Dim node As TreeViewNode = New TreeViewNode()

       node.Text = "Static Tree"

       node.ImagePath = "folders.gif"

       node.Expanded = True

 

 

       Dim childNode As TreeViewNode = New TreeViewNode()

       childNode.Text = "Calendar"

       childNode.ImagePath = "calendar.gif"

 

       node.Items.Add(childNode)

 

       childNode = New TreeViewNode()

       childNode.Text = "Notes"

       childNode.ImagePath = "notes.gif"

 

       node.Items.Add(childNode)

       Return node

End Function
