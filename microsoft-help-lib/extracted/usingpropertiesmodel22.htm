Using Properties Model

The following steps explain how to set the drag preview mode support for nodes through the properties model.

1.   In the controller, create an object for the DiagramPropertiesModel class. Assign this model class to view data. 

 

Controller

 

DiagramPropertiesModel model = new DiagramPropertiesModel();

public ActionResult FlatDiagram()

{

    Node node1 = new Node()

    {

              Name = "Node1",

             Label = "Node1",      

             Shape = Shapes.Rectangle      

    };

    Node node2 = new Node()

    {

             Name = "Node2" ,

            Label = "Node2",      

            Shape = Shapes.Rectangle      

    };

    Node node3 = new Node()

    {

             Name = "Node3", 

             Label = "Node3",      

            Shape = Shapes.Rectangle      

    };

    LineConnector line1 = new LineConnector()

    {

  Name="Line1",

  HeadNode = node1,

  TailNode = node2,

  HeadDecoratorShape = DecoratorShape.Arrow,

  TailDecoratorShape = DecoratorShape.Arrow,

  ConnectorType = ConnectorType.Orthogonal,

  Label="Line1"

    };

    LineConnector line2 = new LineConnector()

    {

  Name = "Line2",

  HeadNode = node1,

  TailNode = node3,

  HeadDecoratorShape = DecoratorShape.Arrow,

  TailDecoratorShape = DecoratorShape.Arrow,

  ConnectorType = ConnectorType.Orthogonal,

  Label = "Line2"

      };

      model.Nodes = new NodesCollection(){ node1, node2, node3 };

      model.Connectors =  new LinesCollection(){ line1, line2 };

      model.Width = 900;

      model.Height = 500;

      model.DiagramMode = DiagramMode.SVG;

      model.AutoFormat = DiagramSkins.Vista;

      model.NodeDraggingMode = DraggingType.PreviewMode;

      ViewData["FlatDiagram"] = model;

      return View();

 } 

 

2.   In the view, invoke the Diagram helper with the view data key as the control ID.

View 

<%{
      Html.Syncfusion().Diagram("FlatDiagram")

          .Render();

  }

%>

 

3.   Build and run the application.
