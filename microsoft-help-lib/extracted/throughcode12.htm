Through Code

 

You can configure the appearance and content of the Syncfusion Menu programmatically. This tutorial shows how to create a menu entirely with code.

 

To create a menu in ASP.NET code:

 

1.   Create a new ASP.NET Web application.

65.  In code view, add the following subroutine to handle the page load event.

 

[C#]

 

using Syncfusion.Web.UI;

using Syncfusion.Web.UI.WebControls.Tools;

using Syncfusion.Web.UI.WebControls.Tools.MenuControl;

 

private void Page_Load(object sender, System.EventArgs e)

{

  if (! IsPostBack)

  {

    BuildBasicMenu();

  }

}

 

/*...............................................*/;

 

private void BuildBasicMenu()

{

  Syncfusion.Web.UI.WebControls.Tools.Menus menu1 = new Menu();

 

  menu1.CustomCSS="css/model.css";

  menu1.ID="menu1";

  menu1.CollapseCascaded=true;

  menu1.StaticPanelCSSClass="Rootpanel";

  menu1.DynamicPanelCSSClass="Panel";

  menu1.CollapseTransition=ETransitionFilter.Pixelate;

  menu1.ExpandTransition=ETransitionFilter.Dissolve;

 

  //Common Look

  MenuItemLook lookcommon = new MenuItemLook();

  lookcommon.ID="lookcommon";

  lookcommon.StateDataDefault.ItemCSSClass="PanelItem";

  lookcommon.StateDataExpanded.ItemCSSClass="PanelItem_Expanded";

  lookcommon.StateDataHover.ItemCSSClass="PanelItem_Hover";

 

  //Root Look

  MenuItemLook lookRoot = new MenuItemLook();

  lookRoot.ID = "lookRoot";

  lookRoot.StateDataDefault.ItemCSSClass = "RootPanelItem";

  lookRoot.StateDataExpanded.ItemCSSClass = "RootPanelItem_Expanded";

  lookRoot.StateDataHover.ItemCSSClass = "RootPanelItem_Hover";

 

  menu1.ItemLooks.Add(lookcommon);

  menu1.ItemLooks.Add(lookRoot);

 

  // Product

  MenuItem product = new MenuItem();

  product.Text="Product";

  product.Look="lookRoot";

 

  //Services

  MenuItem services = new MenuItem();

  services.Text="Services";

  services.Look="lookRoot";

 

  menu1.Items.Add(product);

  menu1.Items.Add(services);

 

  //sub Panel - Services

  MenuItem consult = new MenuItem();

  consult.Text="Consulting";

  consult.Look="lookcommon";

 

  MenuItem fle = new MenuItem();

  fle.Text=".NET Expertise Consultation";

  fle.Look="lookcommon";

  consult.Items.AddItem(fle);

 

  MenuItem train = new MenuItem();

  train.Text="Training";

  train.Look="lookcommon";

  services.Items.AddItem(consult);

  services.Items.AddItem(train);

 

  PlaceHolder1.Controls.Add(menu1);

}//BuildBasicMenu
