Client-Side Events

 

The various client side events of CallbackMultiplexer are given below.

 

 

Event

 

Description

AfterCallbackResponseProcessedScript

Specifies the script to execute after the callback result is processed.

AfterCallbackScript

Specifies the script to execute after the callback request is sent to the server.

BeforeCallbackReponseProcessingScript

Specifies the script to execute before the callback result is processed.

BeforeCallbackScript

Specifies the script to execute before the callback request is sent to the server.

 

Also you can use BeforeCallbackResponseProcessingScript and AfterCallbackResponseProcessedScript client side events of each CallbackPanels, that is refreshed on callback.

 

The following example shows how to use CallbackMultiplexer and CallbackPanels client side events.

 

[ASPX]

 

<table>

    <tr>

        <td>

            <ssw:CallbackPanel ID="CallbackPanel1" runat="server" Height="50px" Width="125px" AfterCallbackResponseProcessedScript="AfterResponseProcess('CallbackPanel1')">

                <asp:Label ID="Label1" Text="CallbackPanel1" runat="server"></asp:Label>

            </ssw:CallbackPanel>

        </td>

        <td>

            <ssw:CallbackPanel ID="CallbackPanel2" runat="server" Height="50px" Width="125px" AfterCallbackResponseProcessedScript="AfterResponseProcess('CallbackPanel2')">

                <asp:Label ID="Label2" Text="CallbackPanel2" runat="server"></asp:Label>

            </ssw:CallbackPanel>

        </td>

    </tr>

</table>

<asp:Button ID="Button1" runat="server" OnClientClick="RefreshPanels(); return false;"

    Text="Refresh Panels" />

<ssw:CallbackMultiplexer ID="CallbackMultiplexer1" ClientObjectId="_sfMultiplexer1" runat="server" OnCallback="CallbackMultiplexer1_Callback" BeforeCallbackScript="BeforeCallback()" AfterCallbackScript="AfterCallback()" BeforeCallbackResponseProcessingScript="BeforeResponseProcess()"/>

 

[JavaScript]

 

function RefreshPanels()

{

    _sfMultiplexer1.callback();

}

function BeforeCallback()

{

    alert("Before callback invoke");

}

function AfterCallback()

{

    alert("Callback invoked");

}

function BeforeResponseProcess()

{

    alert('Start processing script');

}

function AfterResponseProcess( PanelID )

{

    alert( PanelID + ' is refreshed');

}

 

[C#]

 

protected void CallbackMultiplexer1_Callback(object sender, Syncfusion.Web.UI.WebControls.Shared.CallbackEventArgs e)

{

       CallbackPanel1.BackColor = GetRandomBackColor();

       CallbackPanel2.BackColor = GetRandomBackColor();

       CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel1 );

       CallbackMultiplexer1.ControlsToRefresh.Add( CallbackPanel2 );

}

 

private System.Drawing.Color GetRandomBackColor()

{

       System.Threading.Thread.Sleep( 1 );

       Random random = new Random();           

       int nRed = random.Next( 0, 255 );

       int nGreen = random.Next( 0, 255 );

       int nBlue = random.Next( 0, 255 );

       System.Drawing.Color color = System.Drawing.Color.FromArgb( nRed, nGreen, nBlue );

       return color;

}
