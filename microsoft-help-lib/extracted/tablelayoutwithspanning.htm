TableLayout with Spanning

 

The following code example illustrates the spanning options provided by the TableLayout control.

 

[ASPX]

 

<syncfusion:LayoutPanel ID="LayoutPanel1" Title="Table Layout Sample" runat="server" ScrollMode="None" width="1024px" Height="768px">

    <syncfusion:LayoutTable ID="LayoutTable1" runat="server" Width="100%" Height="100%"

        CellSpacing="5">

        <syncfusion:LayoutTableRow ID="LayoutTableRow1" runat="server">

            <syncfusion:LayoutTableCell ID="LayoutTableCell1" RowSpan="3" runat="server">                           

                <syncfusion:LayoutPanel ID="LayoutPanel2" Title="Lots of Spanning" Width="100%" Height="100%"

                    runat="server">

                        <p>

                            Spanning three rows.</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

            <syncfusion:LayoutTableCell ID="LayoutTableCell2" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel3" Title="Basic Table Cell" Width="100%" Height="100%"

                    runat="server">

                        <p>

                            Basic panel in a table cell.</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

            <syncfusion:LayoutTableCell ID="LayoutTableCell3" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel4" ShowHeader="false" Width="100%" Height="100%"

                    runat="server">

                        <p>

                            Plain panel</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

            <syncfusion:LayoutTableCell ID="LayoutTableCell4" RowSpan="2" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel5" Title="Another Cell" Width="100%" Height="100%"

                    runat="server">

                        <p>

                            Spanning two rows.</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

        </syncfusion:LayoutTableRow>

        <syncfusion:LayoutTableRow ID="LayoutTableRow2" runat="server">

            <syncfusion:LayoutTableCell ID="LayoutTableCell5" ColumnSpan="2" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel6" ShowHeader="false" Width="100%" Height="100%"

                    runat="server">

                        <p>

                            Plain cell spanning two columns</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

        </syncfusion:LayoutTableRow>

        <syncfusion:LayoutTableRow ID="LayoutTableRow3" runat="server">

            <syncfusion:LayoutTableCell ID="LayoutTableCell6" ColumnSpan="3" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel7" Title="More Column Spanning" Width="100%"

                    Height="100%" runat="server">

                        <p>

                            Spanning three columns.</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

        </syncfusion:LayoutTableRow>

        <syncfusion:LayoutTableRow ID="LayoutTableRow4" runat="server">

            <syncfusion:LayoutTableCell ID="LayoutTableCell7" ColumnSpan="4" runat="server">

                <syncfusion:LayoutPanel ID="LayoutPanel8" Title="Spanning All Columns" Width="100%"

                    Height="100%" runat="server">

                        <p>

                            Spanning all columns.</p>

                </syncfusion:LayoutPanel>

            </syncfusion:LayoutTableCell>

        </syncfusion:LayoutTableRow>

    </syncfusion:LayoutTable>

</syncfusion:LayoutPanel>
