ToolTip Customization

The ToolTip feature allows users to display ToolTips for chart points.

The ShowTooltips property enables ToolTips for all the chart regions. To render all the chart regions, you need to set the ChartModel.Calcregions property to true.

The ToolTip feature allows users to display ToolTips for the chart points by using PointsTooltipTemplateID formatting.

The ShowTooltips property enables ToolTips for all the chart regions. To render all the chart regions, you need to set the ChartModel.Calcregions property to true.

The ToolTips in the chart can be turned off by using the control's ShowToolTips property.

The steps to implement rich ToolTip support for the chart series are as follows.

 

1.   Set the ShowTooltips property of the chart model to True and set the ChartModel.Calcregions property to True.

2.   Set the PointsTooltipTemplateID property of the chart model for custom string format.

3.   Set the ChartSeries, ChartArea, and Chart properties.

 

 Rich ToolTips can be added to the series using the following code.

 

[C#]     

Series series = new Series("CompanyA");

series.Type = SeriesType.Column;

series.BindTo.DataSource = this.GetData();

series.BindTo.XName="EmployeeID";

series.BindTo.YNames=new string[]{"Rating"};

 

this.ChartAdv1.Series.Add(series);

 

this.ChartAdv1.CalcRegion=true;

this.ChartAdv1.ShowToolTip=true;

series.PointsTooltipTemplateID = "Tooltip";

 

this.ChartAdv1.ClientSideEvents.OnLoad="OnLoad";

this.ChartAdv1.ClientSideEvents.OnPointRegionMouseMove="OnMouseMove";

 

public List<Employee> GetData()

    {

        ExcelEngine engine = new ExcelEngine();

        IApplication app = engine.Excel;

        IWorkbook workBook = app.Workbooks.Open(Request.PhysicalApplicationPath +@"App_Data\ChartData.xlsx");

        IWorksheet sheet = workBook.Worksheets[1];

        DataTable dt = sheet.ExportDataTable(sheet.UsedRange, ExcelExportDataTableOptions.DetectColumnTypes);

        List<Employee> data = new List<Employee>();

 

        for (int i = 1; i < 7; i++)

        {

            Employee emp = new Employee();

            emp.EmployeeID = Convert.ToInt32(dt.Rows[i][0]);

            emp.FirstName = Convert.ToString(dt.Rows[i][1]);

            emp.LastName = Convert.ToString(dt.Rows[i][2]);

            emp.Title = Convert.ToString(dt.Rows[i][3]);

            emp.Country = Convert.ToString(dt.Rows[i][4]);

            emp.Rating = Convert.ToInt32(dt.Rows[i][5]);

            data.Add(emp);

        }

 

        return data;

 

    }

Public Function GetData() As List(Of Employee)

        Dim engine As ExcelEngine = New ExcelEngine()

        Dim app As IApplication = engine.Excel

        Dim workBook As IWorkbook = app.Workbooks.Open(Request.PhysicalApplicationPath & "App_Data\ChartData.xlsx")

        Dim sheet As IWorksheet = workBook.Worksheets(1)

        Dim dt As DataTable = sheet.ExportDataTable(sheet.UsedRange, ExcelExportDataTableOptions.DetectColumnTypes)

        Dim data As List(Of Employee) = New List(Of Employee)()

 

        For i As Integer = 1 To 6

            Dim emp As Employee = New Employee()

            emp.EmployeeID = Convert.ToInt32(dt.Rows(i)(0))

            emp.FirstName = Convert.ToString(dt.Rows(i)(1))

            emp.LastName = Convert.ToString(dt.Rows(i)(2))

            emp.Title = Convert.ToString(dt.Rows(i)(3))

            emp.Country = Convert.ToString(dt.Rows(i)(4))

            emp.Rating = Convert.ToInt32(dt.Rows(i)(5))

            data.Add(emp)

        Next i

 

        Return data

 

    End Function

[JavaScript]

 

 <script type="text/javascript">

        var chartModel = null;

        $(document).ready(function () {

            if (!$.browser.msie)

                $("#Tooltip").find("tr:first").remove();

            $(document).bind("mousemove", function (e) {

                if ($(e.target).hasClass("tooltipDiv") || $(e.target).parents(".tooltipDiv").length > 0) {

                    $(".tooltipDiv").remove();

                }

            });

        });

        function OnLoad(sender, args) {

            chartModel = sender.model;

        }

        function OnMouseMove(sender, args) {

            var series = chartModel.Series[args.Data.Region.SeriesIndex];

            var point = series.Points[args.Data.Region.Region.PointIndex];

            var data = $.Enumerable.From(series.BindTo.DataSource).Where(function (p) {

                return point.X == p.EmployeeID;

            }).ToArray();

 

            $(".FirstName").html(data[0].FirstName);

            $(".LastName").html(data[0].LastName);

            $(".Title").html(data[0].Title);

            $(".Country").html(data[0].Country);

            $(".Rating").html(data[0].Rating);

 

        }

    </script>
