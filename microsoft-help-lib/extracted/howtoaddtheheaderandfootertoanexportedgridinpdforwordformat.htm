How to add the header and footer to an exported Grid in PDF or Word format?

To add Header and Footer to a Grid that was exported to Word and (or) PDF, use the constructors and properties as shown in the below code snippets.

You can use either code in C# or VB for this purpose, as shown below:

For PDF

Using C#

[C#]

 

GridPdfExport pdf = new GridPdfExport(this.GridGroupingControl1, "PdfExport.pdf", ConverterOptionsExt.All , true, true);

 

(or)

 

GridPdfExport pdf = new GridPdfExport(this.GridGroupingControl1, "PdfExport.pdf", ConverterOptionsExt.All);

pdf.ShowHeader= true;

pdf.ShowFooter= true;

 

 

pdf.DrawPDFHeader += new DrawPDFHeaderFooterEventHandler(pdf_DrawPDFHeader);

pdf.DrawPDFFooter += new DrawPDFHeaderFooterEventHandler(pdf_DrawPDFFooter);

 

pdf.Export();

 

void pdf_DrawPDFHeader(object sender, PDFHeaderFooterEventArgs e)

{

PdfPageTemplateElement header = e.HeaderFooterTemplate;

 

PdfFont font = new PdfStandardFont(PdfFontFamily.Helvetica, 24);

float doubleHeight = font.Height * 2;

Color activeColor = Color.FromArgb(44, 71, 120);

SizeF imageSize = new SizeF(110f, 35f);

//Locating the logo on the right corner of the Drawing Surface

PointF imageLocation = new PointF(e.HeaderFooterTemplate.Width - imageSize.Width - 20, 5);

 

PdfSolidBrush brush = new PdfSolidBrush(activeColor);

 

PdfPen pen = new PdfPen(Color.DarkBlue, 3f);

font = new PdfStandardFont(PdfFontFamily.Helvetica, 16, PdfFontStyle.Bold);

 

//Set formattings for the text

PdfStringFormat format = new PdfStringFormat();

format.Alignment = PdfTextAlignment.Center;

format.LineAlignment = PdfVerticalAlignment.Middle;

 

 

//Draw title

header.Graphics.DrawString("Syncfusion Essential Grid", font, brush, new RectangleF(0, 0, header.Width, header.Height), format);

brush = new PdfSolidBrush(Color.Gray);

font = new PdfStandardFont(PdfFontFamily.Helvetica, 6, PdfFontStyle.Bold);

 

format = new PdfStringFormat();

format.Alignment = PdfTextAlignment.Left;

format.LineAlignment = PdfVerticalAlignment.Bottom;

 

//Draw some lines in the header

pen = new PdfPen(Color.DarkBlue, 0.7f);

header.Graphics.DrawLine(pen, 0, 0, header.Width, 0);

pen = new PdfPen(Color.DarkBlue, 2f);

header.Graphics.DrawLine(pen, 0, 03, header.Width + 3, 03);

pen = new PdfPen(Color.DarkBlue, 2f);

header.Graphics.DrawLine(pen, 0, header.Height - 3, header.Width, header.Height - 3);

header.Graphics.DrawLine(pen, 0, header.Height, header.Width, header.Height);

}

 

void pdf_DrawPDFFooter(object sender, PDFHeaderFooterEventArgs e)

{

PdfPageTemplateElement footer = e.HeaderFooterTemplate;

 

PdfFont font = new PdfStandardFont(PdfFontFamily.Helvetica, 8);

 

PdfSolidBrush brush = new PdfSolidBrush(Color.Gray);

 

PdfPen pen = new PdfPen(Color.DarkBlue, 3f);

font = new PdfStandardFont(PdfFontFamily.Helvetica, 6, PdfFontStyle.Bold);

PdfStringFormat format = new PdfStringFormat();

format.Alignment = PdfTextAlignment.Center;

format.LineAlignment = PdfVerticalAlignment.Middle;

footer.Graphics.DrawString("@Copyright 2011", font, brush, new RectangleF(0, footer.Height - 40, footer.Width, footer.Height), format);

 

format = new PdfStringFormat();

format.Alignment = PdfTextAlignment.Right;

format.LineAlignment = PdfVerticalAlignment.Bottom;

 

//Create page number field

PdfPageNumberField pageNumber = new PdfPageNumberField(font, brush);

 

//Create page count field

PdfPageCountField count = new PdfPageCountField(font, brush);

 

PdfCompositeField compositeField = new PdfCompositeField(font, brush, "Page {0} of {1}", pageNumber, count);

compositeField.Bounds = footer.Bounds;

compositeField.Draw(footer.Graphics, new PointF(470, footer.Height - 10));

 

}

 

 

 

Using VB

[VB]

Dim pdf As GridPdfExport

pdf = New GridPdfExport(Me.GridGroupingControl1, "PdfExport.pdf", ConverterOptionsExt.All, true, true)

 

(or)

 

Dim pdf As GridPdfExport = New GridPdfExport(Me.GridGroupingControl1, "PdfExport.pdf", ConverterOptionsExt.All)

pdf.ShowFooter= True

 

AddHandler pdf.DrawPDFHeader, AddressOf pdf_DrawPDFHeader

For Word

Using C#

[C#]

GridWordExport word = new GridWordExport(this.GridGroupingControl1, "WordDoc.docx", ConverterOptionsExt.All, true, true)

 

(or)

 

GridWordExport word = new GridWordExport(this.GridGroupingControl1, "WordDoc.docx", ConverterOptionsExt.All)

word.ShowHeader= true;

word.ShowFooter= true;

 

 

word.DrawHeader += new DrawDocHeaderFooterEventHandler(word_DrawHeader);

word.DrawFooter += new DrawDocHeaderFooterEventHandler(word_DrawFooter);

 

word.Export();

 

void word_DrawFooter(object sender, DocHeaderFooterEventArgs e)

{

IWTextRange txt = e.Footer.AddParagraph().AppendText("\t\t\t\t\t\tCopyright Syncfusion Inc. 2001 - 20011");

txt.CharacterFormat.Font = new Font("verdana", 12f, FontStyle.Bold);

}

 

void word_DrawHeader(object sender, DocHeaderFooterEventArgs e)

{

IWTextRange txt = e.Header.AddParagraph().AppendText("\t\t\t\t\t\t\tEssential Grid\n");

txt.CharacterFormat.Font = new Font("verdana", 12f, FontStyle.Bold);

}

 

 

 

Using VB

[VB]

Dim word As GridWordExport = New GridWordExport(Me.GridGroupingControl1, "WordDoc.docx", ConverterOptionsExt.All, True, True)

 

(Or)

Dim word As GridWordExport = New GridWordExport(Me.GridGroupingControl1, "WordDoc.docx", ConverterOptionsExt.All)

 

word.ShowHeader= true;

word.ShowFooter= true;

 

 

Private word.DrawHeader += New DrawDocHeaderFooterEventHandler(AddressOf word_DrawHeader)

Private word.DrawFooter += New DrawDocHeaderFooterEventHandler(AddressOf word_DrawFooter)

 

word.Export()

 

 

Private Sub word_DrawFooter(ByVal sender As Object, ByVal e As DocHeaderFooterEventArgs)

 

Dim txt As IWTextRange = e.Footer.AddParagraph().AppendText(Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab & "Copyright Syncfusion Inc. 2001 - 20011")

txt.CharacterFormat.Font = New Font("verdana", 12.0F, FontStyle.Bold)

End Sub

 

Private Sub word_DrawHeader(ByVal sender As Object, ByVal e As DocHeaderFooterEventArgs)

 

Dim txt As IWTextRange = e.Header.AddParagraph().AppendText(Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab + Constants.vbTab & "Essential Grid" & Constants.vbLf)

txt.CharacterFormat.Font = New Font("verdana", 12.0F, FontStyle.Bold)

 

End Sub
