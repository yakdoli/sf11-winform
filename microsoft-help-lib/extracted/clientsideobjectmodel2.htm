Return Type

Sample demonstrates how to show popup when user clicks on the html text element or edits the text.

 

[ASPX]

 

<ssw:GenericDropDown onclick="_sfDropDown1.ShowPopup()" onkeydown="_sfDropDown1.ShowPopup()" ClientObjectId="_sfDropDown1" ID="DropDown1" runat="server" Width="200px" PopupHeight="100px" PopupWidth="200px" AutoFormat="Office2007 Luna Blue">           

    <table style="width:100%; height:100%">

        <tr>

            <td>

                some popup content

            </td>

            <td valign="top" align="right">

                <input type="button" value="hide popup" onclick="_sfDropDown1.HidePopup()" />

            </td>

        </tr>

    </table>

</ssw:GenericDropDown>

GenericDropDownOnTextChanged

 

[javascript]

 

function GenericDropDownOnTextChanged( oEventData )

{

    var sMessage = "Text '" + oEventData.OldText + "' was changed by '" + oEventData.Text + "'";

    alert( sMessage );

}
