Data Binding

The CheckedListBox control provides a rich DataBinding support that allows users to fill the CheckedListBox items using Collections.

 

Use Case Scenarios

This Feature will be useful when you want to bind the data from the collections or from the database by setting the ItemsSource property of the CheckeddListBox as Collections or DataSet.Not applicable.

 

Adding Data Binding Feature to an Application

The following code example explains the addition of the Data Binding feature to the application.

 

[XAML]

<syncfusion:CheckedListBox Grid.Row="0" x:Name="checklistbox" Mode="Checked"  BorderThickness="1"  CheckedListBoxAlignment="Left"   ItemsSource="{Binding Countries}"  DisplayMemberPath="Name"       CheckOnClick="False"    Height="225" Width="240" >                     

</syncfusion:CheckedListBox>              

 

[C#]

 

public partial class MainPage : PhoneApplicationPage

    {

 

        public ObservableCollection<Country> Countries

        {

            get { return (ObservableCollection<Country>)GetValue(CountriesProperty); }

            set { SetValue(CountriesProperty, value); }

        }

 

        // Using a DependencyProperty as the backing store for Countries.  This enables animation, styling, binding, etc...

        public static readonly DependencyProperty CountriesProperty =

            DependencyProperty.Register("Countries", typeof(ObservableCollection<Country>), typeof(MainPage), new PropertyMetadata(null));

 

        // Constructor

        public MainPage()

        {

          

 

            InitializeComponent();

            DataContext = this;

 

            this.Countries = new ObservableCollection<Country>();

            this.Countries.Add(new Country()

            {

                Name = "United Kindom",

                Code = "UK"

            });

            this.Countries.Add(new Country()

            {

                Name = "Canada",

                Code = "CA"

            });

 

            this.Countries.Add(new Country()

            {

                Name = "Germany",

                Code = "DE"

            });

            this.Countries.Add(new Country()

            {

                Name = "India",

                Code = "IN"

            });

            this.Countries.Add(new Country()

            {

                Name = "Ukraine",

                Code = "UA"

            });

        }

    }

 

    public class Country

    {

        #region Properties

        public string Name { get; set; }

        public string Code { get; set; }

        #endregion

    }
