Creating the Diagram using DiagramBuilder

To create the Diagram control using view customization:

1.   Right-click the View >Home folder.

2.   Click Add, and then select View.

3.   Name the FlatDiagram and click the Add button.

 

Figure 22: Add View Dialog Box

 

4.   Add the following code in the FlatDiagram.aspx file to create the Diagram control in the view:

 

View

 

<% Node node1 = new Node()

   {

       Name = "Node1",

       Label = "Node1",

       LabelHorizontalAlignment = Horizontal.Center,

       LabelVerticalAlignment = Vertical.Middle,

       OffsetX = 225,

       OffsetY = 100,

       Shape = Shapes.Ellipse,

       Height = 50.00,

       Width = 50.00,

       BackgroundColor = "orange",

       BorderColor = "black",

       BorderWidth = 1,

       LabelFontColor = "black",

       LabelFontSize = 12,

       LabelFontFamily="Times New Roman"      

   };

   Node node2 = new Node()

   {

       Name = "Node2",

       Label = "Node2",

       LabelHorizontalAlignment = Horizontal.Center,

       LabelVerticalAlignment = Vertical.Middle,

       OffsetX = 100.00,

       OffsetY = 300.00,

       Shape = Shapes.Rectangle,

       Height = 50.00,

       Width = 100.00,

       BackgroundColor = "orange",

       BorderColor = "black",

       LabelBorderWidth = 1,

       LabelFontColor = "black",

       LabelFontSize = 12,

       LabelFontFamily="Times New Roman"

   };

   Node node3 = new Node()

   {

       Name = "Node3",

       Label = "Node3",

       LabelHorizontalAlignment = Horizontal.Center,

       LabelVerticalAlignment = Vertical.Middle,

       OffsetX = 300.00,

       OffsetY = 300.00,

       Shape = Shapes.Rectangle,

       Height = 50.00,

       Width = 100.00,

       BackgroundColor = "orange",

       BorderColor = "black",

       LabelBorderWidth = 1,

       LabelFontColor = "black",

       LabelFontSize = 12,

       LabelFontFamily="Times New Roman"      

   };

 

LineConnector line1 = new LineConnector()

   {

       Name= "line1",

       HeadNode = node1,

       TailNode = node2,

       ConnectorType = ConnectorType.Straight,

       LineColor = "black",

       LineWidth = 1

   };

   LineConnector line2 = new LineConnector()

   {

       Name= "line2",

       HeadNode = node1,

       TailNode = node3,

       ConnectorType = ConnectorType. Straight,

       LineColor = "black",

       LineWidth = 1

   };

%>

<%{

  Html.Syncfusion().Diagram("FlatDiagram")

          .Nodes(nodes => nodes.Add(node1).Add(node2).Add(node3))

          .Connectors(conn => conn.Add(line1).Add(line2))             

          .Width(Unit.Pixel(1100))

          .Height(Unit.Pixel(500))

           .DiagramMode(DiagramMode.SVG)

          .Render();

  }

%>

 

Note: If you want to create the diagram in the Canvas mode. Change the DiagramMode to Canvas. By default the diagram is rendered in the SVG mode.

 

5.   Open the ~/Controllers/HomeController.cs.

6.   Include the following namespaces in the HomeController:

a.   Syncfusion.Mvc.Diagram

b.   Syncfusion.Mvc.Shared

 

7.   Include the Syncfusion.Mvc.Shared and Syncfusion.Mvc.Diagram namespaces to the HomeController by using the following code:

 

Controller

using Syncfusion.Mvc.Diagram;

using Syncfusion.Mvc.Shared;

 

 

8.   The controller page looks like this:

 

Controller

 

public ActionResult FlatDiagram()

{

    return View();

}

 

 

 

 

9.   Run the application.
