On Demand Paging

 

This section explains the paging concept on demand.

 

[C#]

 

using Syncfusion.Web.UI.WebControls.Grid.Grouping;

 

public partial class OnDemandPaging : System.Web.UI.Page {

    protected void Page_Load( object sender, EventArgs e ) {

        if ( !IsPostBack )

        {

            this.GridGroupingControl1.DataSource = GetData( );

            ( ( IPagingContainer )this.GridGroupingControl1 ).IsPagingOnDemand = true;

        }

        ( ( IPagingContainer )this.GridGroupingControl1 ).OnDemandDataSourceLoad += new EventHandler<GridPagerOnDemandDataSourceLoadArgs>( OnDemandPaging_OnDemandDataSourceLoad );

        ( ( IPagingContainer )this.GridGroupingControl1 ).OnDemandTotalRowCount += new EventHandler<GridPagerOnDemandTotalRowCountArgs>( OnDemandPaging_OnDemandTotalRowCount );

        this.GridGroupingControl1.QueryCellStyleInfo += new GridTableCellStyleInfoEventHandler( GridGroupingControl1_QueryCellStyleInfo );

    }

 

    void GridGroupingControl1_QueryCellStyleInfo( object sender, GridTableCellStyleInfoEventArgs e ) {

        e.Style.WrapText = false;

    }

 

    void OnDemandPaging_OnDemandTotalRowCount( object sender, GridPagerOnDemandTotalRowCountArgs e ) {

        e.TotalRowCount = 100;

    }

 

    protected override bool OnBubbleEvent( object source, EventArgs args ) {

        return base.OnBubbleEvent( source, args );

    }

 

    void OnDemandPaging_OnDemandDataSourceLoad( object sender, GridPagerOnDemandDataSourceLoadArgs e ) {

        //provide the paged data source here. Can use LINQ queries to get the paged records from the database

        e.PagedDataSource = GetFilteredTable( ( ( DataTable )this.Session["Table"] ), e.PagedRows, e.PagedRows + e.MaximumRows );

    }

 

    private DataTable GetData( ) {

        DataSet ds = new DataSet( );

        using ( OleDbConnection oConn = new OleDbConnection( System.Configuration.ConfigurationManager.ConnectionStrings["ConnectionString"].ConnectionString ) )

        {

            OleDbDataAdapter oDa = new OleDbDataAdapter( "SELECT * FROM Orders", oConn );

            oDa.Fill( ds );

        }

        this.Session["Table"] = ds.Tables[0];

        return GetFilteredTable( ds.Tables[0], 0, 12 );

    }

 

    private DataTable GetFilteredTable( DataTable dt, int startIdx, int rowCount ) {

        DataTable cloneTable = dt.Clone( );

        for ( int i = startIdx; i < rowCount; i++ )

        {

            cloneTable.ImportRow( dt.Rows[i] );

        }

        return cloneTable;

    }

}

 

[VB.NET]

 

Imports Syncfusion.Web.UI.WebControls.Grid.Grouping

 

Partial Public Class OnDemandPaging : Inherits System.Web.UI.Page

Protected Sub Page_Load(ByVal sender As Object, ByVal e As EventArgs)

            If (Not IsPostBack) Then

                Me.GridGroupingControl1.DataSource = GetData()

                CType(Me.GridGroupingControl1, Syncfusion.Web.UI.WebControls.Grid.Grouping.IPagingContainer).IsPagingOnDemand = True

            End If

            AddHandler (CType(GridGroupingControl1, IPagingContainer)).OnDemandDataSourceLoad, AddressOf Of GridPagerOnDemandDataSourceLoadArgs

            AddHandler (CType(GridGroupingControl1, IPagingContainer)).OnDemandTotalRowCount, AddressOf Of GridPagerOnDemandTotalRowCountArgs

            Me.GridGroupingControl1.QueryCellStyleInfo += New Syncfusion.Web.UI.WebControls.Grid.Grouping.GridTableCellStyleInfoEventHandler(GridGroupingControl1_QueryCellStyleInfo)

End Sub

 

Private Sub GridGroupingControl1_QueryCellStyleInfo(ByVal sender As Object, ByVal e As Syncfusion.Web.UI.WebControls.Grid.Grouping.GridTableCellStyleInfoEventArgs)

            e.Style.WrapText = False

End Sub

 

Private Sub OnDemandPaging_OnDemandTotalRowCount(ByVal sender As Object, ByVal e As Syncfusion.Web.UI.WebControls.Grid.Grouping.GridPagerOnDemandTotalRowCountArgs)

            e.TotalRowCount = 100

End Sub

 

Protected Overrides Function OnBubbleEvent(ByVal source As Object, ByVal args As EventArgs) As Boolean

            Return MyBase.OnBubbleEvent(source, args)

End Function

 

Private Sub OnDemandPaging_OnDemandDataSourceLoad(ByVal sender As Object, ByVal e As Syncfusion.Web.UI.WebControls.Grid.Grouping.GridPagerOnDemandDataSourceLoadArgs)

            'provide the paged data source here. Can use LINQ queries to get the paged records from the database

            e.PagedDataSource = GetFilteredTable((CType(Me.Session("Table"), Data.DataTable)), e.PagedRows, e.PagedRows + e.MaximumRows)

End Sub

 

Private Function GetData() As Data.DataTable

            Dim ds As Data.DataSet = New Data.DataSet()

            Using oConn As Data.OleDb.OleDbConnection = New Data.OleDb.OleDbConnection(System.Configuration.ConfigurationManager.ConnectionStrings("ConnectionString").ConnectionString)

                Dim oDa As Data.OleDb.OleDbDataAdapter = New Data.OleDb.OleDbDataAdapter("SELECT * FROM Orders", oConn)

                oDa.Fill(ds)

            End Using

            Me.Session("Table") = ds.Tables(0)

            Return GetFilteredTable(ds.Tables(0), 0, 12)

End Function

 

Private Function GetFilteredTable(ByVal dt As Data.DataTable, ByVal startIdx As Integer, ByVal rowCount As Integer) As Data.DataTable

            Dim cloneTable As Data.DataTable = dt.Clone()

            Dim i As Integer = startIdx

            Do While i < rowCount

                cloneTable.ImportRow(dt.Rows(i))

                i += 1

            Loop

            Return cloneTable

End Function

End Class
