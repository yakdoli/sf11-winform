Custom Label

This feature allows you to add custom text labels using the CustomLabel element of gauge. 

The label value can be set using its  LabelValue  property. You can also customize the custom text, using its FontSize and FontFamily properties.

 The location of the label can be customized using its Location property. The custom text angle can be set using its TextAngle  property.

The code snippets for adding scale to the circular gauge are given below:

 

[ASPX]

 

< syncfusion : LinearGauge FrameType ="RoundedRectangle" AutoFormat ="Blend" ID ="LinearGauge1"

            runat="server">

            <Scales>

                <syncfusion:LinearScaleScaleBarSize="20"MajorIntervalValue="10"MinorIntervalValue="2">

                    <Labels>

                        <syncfusion:LinearGaugeLabelLabelPlacement="Near"LabelStyle="MajorInterval"XDistanceFromScale="-10"/>

                    </Labels>

                    <Ticks>

                        <syncfusion:LinearGaugeTickTickPlacement="Near"TickStyle="MajorInterval"TickHeight="10"/>

                        <syncfusion:LinearGaugeTickTickPlacement="Near"TickStyle="MinorInterval"TickHeight="5"/>

                    </Ticks>

                    <BarPointers>

                        <syncfusion:LinearBarPointerPointerWidth="20"Value="50"/>

                    </BarPointers>

                    <MarkerPointers>

                        <syncfusion:LinearMarkerPointerMarkerStyle="Diamond"Value="50"PointerPlacement="Far"/>

                    </MarkerPointers>

                    <CustomLabel>

                    <syncfusion:GaugeCustomLabelLabelValue="Gauge"Location="50, 97"/>

                    </CustomLabel>

                </syncfusion:LinearScale>

            </Scales>

        </syncfusion:LinearGauge>

 

[C#]

 

protected void Page_Load(object sender, EventArgs e)

    {

        BuildLinearGauge();

    }

    privatevoid BuildLinearGauge()

    {

        LinearGauge gauge1 = newLinearGauge();

        gauge1.FrameType = LinearGaugeFrameType.RoundedRectangle;

        gauge1.AutoFormat = GaugeSkins.Blend;

 

        LinearScale scale1 = newLinearScale();

        scale1.ScaleBarSize = 20;

        scale1.MajorIntervalValue = 10;

        scale1.MinorIntervalValue = 2;

        gauge1.Scales.Add(scale1);

 

        LinearGaugeLabel label1 = newLinearGaugeLabel();

        label1.LabelPlacement = ScalePlacement.Near;

        label1.XDistanceFromScale = -10;

        scale1.Labels.Add(label1);

 

        LinearGaugeTick tick1 = newLinearGaugeTick();

        tick2.TickHeight = 10;

        scale1.Ticks.Add(tick1);

 

        LinearGaugeTick tick2 = newLinearGaugeTick();

        tick2.TickHeight = 5;

        tick2.TickStyle = TickStyle.MinorInterval;

        scale1.Ticks.Add(tick2);

 

        LinearBarPointer barPointer1 = newLinearBarPointer();

        barPointer1.Value = 50;

        scale1.BarPointers.Add(barPointer1);

 

        LinearMarkerPointer markerPointer1 = newLinearMarkerPointer();

        markerPointer1.Value = 50;

        markerPointer1.MarkerStyle = LinearMarkerStyle.Diamond;

        markerPointer1.PointerPlacement = ScalePlacement.Far;

        scale1.MarkerPointers.Add(markerPointer1);

 

        GaugeCustomLabel customLabel = newGaugeCustomLabel();

        customLabel.LabelValue = "Gauge";

        customLabel.Font = new         Syncfusion.Web.UI.WebControls.Gauge.Font("Arial", "13", FontStyle.Bold);

        customLabel.Location = newPoint(50, 97);

        scale1.CustomLabel.Add(customLabel);

 

        this.form1.Controls.Add(gauge1);

}
