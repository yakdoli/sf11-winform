Binding to LinqDataSource

 

This section illustrates how to bind LinqDataSource with the Schedule control.

 

1.   Set Data Source control to bind data to the Schedule Control.

The below example demonstrates the resulting code on setting the LinqDataSource control as data source to the Schedule control.

 

[ASPX]

 

<asp:UpdatePanel ID="UpdatePanel1" runat="server">

            <ContentTemplate>

                <syncfusion:Schedule ID="Schedule1" runat="server" AutoFormat="Office2007Blue" ScheduleType="Day"

                    StartDate="2008-08-28" Width="900px" AutoPostBackOnScheduleClick="true" ShowContextMenu="true"

                    AppointmentDataSourceID="LinqDataSource1" CategoryDataSourceID="LinqDataSource2"

                    ResourceDataSourceID="LinqDataSource3" ViewStrip="true" Height="500px" ResolutionHeight="16"

                    DisabledBackgroundColor="221, 227, 238" WaitingPopupCSSClass="PopupCSS">

                    <ResourceColumn Width="50px" />

<AppointmentBindProperties AllDayField="AllDay" WeekDayField="WeekDay" MonthNumberField="MonthNumber"

                        MonthDateValueField="MonthDateValue" EndReccurenceField="EndRecurrence" StartReccurenceField="StartRecurrence"

                        OccurrencesNumberField="OccurrenceNumber" ReccurencePatternField="RecurrencePattern"

                        UseFridayField="UseFriday" UseSundayField="UseSunday" UseTuesdayField="UseTuesday"

                        UseWednesdayField="UseWednesday" UseThursdayField="UseThursday" UseSaturdayField="UseSaturday"  FirstConditionField="FirstCondition" UseMondayField="UseMonday" AllowRecurrenceField="AllowRecurrence"

                        ContentField="Content" SubjectField="Subject" UniqueIDField="Id" EndTimeField="EndTime"

                        LocationValueField="LocationValue" StartTimeField="StartTime" OwnerField="Owner" />                </syncfusion:Schedule>

            </ContentTemplate>

        </asp:UpdatePanel>

        <asp:LinqDataSource ID="LinqDataSource1" EnableUpdate="true" EnableInsert="true" EnableDelete="true" ContextTypeName="ScheduleDataDataContext" TableName="AppointmentsTables"

            runat="server">

        </asp:LinqDataSource>

        <asp:LinqDataSource ID="LinqDataSource2" EnableUpdate="true" EnableInsert="true"

            EnableDelete="true" ContextTypeName="ScheduleDataDataContext" TableName="CategoriesTables"

            runat="server">

        </asp:LinqDataSource>

        <asp:LinqDataSource ID="LinqDataSource3" EnableUpdate="true" EnableInsert="true"

            EnableDelete="true" ContextTypeName="ScheduleDataDataContext" TableName="ResourceTables"

            runat="server">

        </asp:LinqDataSource>
