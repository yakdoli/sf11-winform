Adding Controls into the Panel

 

You can add any html or web controls into CallbackPanel control through designer or code.

 

You can drag from the Toolbox, any html or web controls to CallbackPanel at design time and also edit these controls. If necessary, the panel will resize itself to accommodate the control.

 

Figure 163

 

You can create any controls programmatically and add them to the Controls collection of existing CallbackPanel control. Also, you can use Tag property of CallbackPanel to attach any object to this control's view state.

 

To add controls into CallbackPanel on page callback see the following code example.

 

[ASPX]

 

<ssw:CallbackPanel ID="CallbackPanel1" runat="server" OnCallbackRefresh="CallbackPanel1_CallbackRefresh">

    <asp:Button ID="ButtonAdd" Text="Add Label" runat="server" OnClientClick="_sfCallbackPanel1.callback('add');return false" />

    <asp:Button ID="ButtonRemove" Text="Remove Label" runat="server" OnClientClick="_sfCallbackPanel1.callback('remove');return false" />

</ssw:CallbackPanel>

 

[C#]

 

protected void Page_Load(object sender, EventArgs e)

{

       if( !Page.IsPostBack )

       {

              CallbackPanel1.Tag = 0;

              ButtonRemove.Enabled = false;

       }

       int numlabels = (int)CallbackPanel1.Tag;

       for( int i= 1; i <= numlabels; i++ )

       {

              AddLabel( i );

       }

}

 

protected void CallbackPanel1_CallbackRefresh(object sender, CancellableCallbackEventArgs e)

{

       int nLabelsCount = (int)CallbackPanel1.Tag;

       switch(e.CallbackArgument )

       {

              case "add":

                     nLabelsCount++;                         

                     AddLabel( nLabelsCount );

                     break;

              case "remove":

                     nLabelsCount--;

                     //Remove last Label control

                     CallbackPanel1.Controls.RemoveAt( CallbackPanel1.Controls.Count - 1 );

                     break;

       }

       ButtonRemove.Enabled = 0 < nLabelsCount;

       // Save labels count in Tag property of CallbackPanel

       CallbackPanel1.Tag = nLabelsCount;

}

 

private void AddLabel( int nIndex )

{

       Label myLabel = new Label();

 

       // Set the label's Text and ID properties.

       myLabel.Text = "<br/>" + "Label" + nIndex;

       myLabel.ID = "Label" + nIndex;

 

       //add Label control to CallbackPanel control

       CallbackPanel1.Controls.Add( myLabel );

}
