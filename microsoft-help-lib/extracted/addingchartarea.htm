     Adding Chart Area

 

You can add a chart area object with series data, to the Chart easily, using the following code snippets.

 

[XAML]

 

<UserControl.Resources>

                <!--Initialize chart series color-->

                <LinearGradientBrush x:Key="SeriesColor" EndPoint="1,0.5" StartPoint="0,0.5">

                    <LinearGradientBrush.GradientStops>

                        <GradientStop Color="#FF1C3D6A" Offset="0"/>

                        <GradientStop Color="#FF345580" Offset="0.164841"/>

                        <GradientStop Color="#FF7AAEDE" Offset="0.642853"/>

                        <GradientStop Color="#FF183C7D" Offset="1"/>

                    </LinearGradientBrush.GradientStops>

                </LinearGradientBrush>

            </UserControl.Resources>

 

            <syncfusion:Chart>

                <syncfusion:ChartArea Background="Orange">

                    <syncfusion:ChartSeries Data="1,7,2,10" Interior="{StaticResource SeriesColor}"/>

                </syncfusion:ChartArea>

            </syncfusion:Chart>

 

[C#]

 

Chart chart = new Chart();

ChartArea area = new ChartArea();

 

//Initialize the ChartSeries and add it into the Chart Area.

ChartSeries series = new ChartSeries();

series.Interior=this.Resources["SeriesColor"] as LinearGradientBrush

area.Series.Add(series);

 

//Adding Chart Area into the Chart control

chart.Areas.Add(area);
