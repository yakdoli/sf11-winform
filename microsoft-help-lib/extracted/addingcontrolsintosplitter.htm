Adding controls into Splitter

 

You can add any html or web controls into SplitPane or SlidingPane controls through designer or code.

 

You can drag any html or web controls from toolbox, to the SplitPane or SlidingPane at design time and also edit these controls.

 

Figure 388: Splitter

 

You can create any controls programmatically and add them to the Controls collection of existing SplitPane or SlidingPane controls. Use the below code to do this.

 

[ASPX]

 

<cc1:Splitter ID="Splitter1" runat="server" Height="200px" Width="400px">

    <cc1:SplitPane ID="SplitPane1" runat="server" Width="23px">

        <cc1:SlidingZone ID="SlidingZone1" runat="server" SlideDirection="right">

            <cc1:SlidingPane ID="SlidingPane1" runat="server" Width="150px" Title="TreeView">

            </cc1:SlidingPane>

            <cc1:SlidingPane ID="SlidingPane2" runat="server" Title="Calendar" Width="200px">

            </cc1:SlidingPane>

        </cc1:SlidingZone>

    </cc1:SplitPane>

    <cc1:SplitterBar ID="SplitterBar1" runat="server" CollapseMode="forward" />

    <cc1:SplitPane ID="SplitPane2" runat="server">

    </cc1:SplitPane>

</cc1:Splitter>

 

[C#]

 

using System;

using System.Web.UI.WebControls;

 

public partial class AddingControlsIntoSplitter : System.Web.UI.Page

{

    protected void Page_Load(object sender, EventArgs e)

    {

            /*Add ASP TreeView to SlidingPane1*/

            TreeView TreeView1 = new TreeView();

            TreeView1.ID = "TreeView1";

            for(int i = 1; i < 4; i++ )

            {

                  TreeNode nodeRoot = new TreeNode( "RootNode" + i );

                  TreeView1.Nodes.Add( nodeRoot );

                  for( int j = 1; j< 4; j++ )

                  {

                        TreeNode nodeChild = new TreeNode( "Child Node" + j );

                        nodeRoot.ChildNodes.Add( nodeChild );

                  }

            }

            TreeView1.ImageSet = TreeViewImageSet.Faq;

            TreeView1.Font.Size = 10;

           

            SlidingPane1.Controls.Add( TreeView2 );

           

            /*Add ASP Calendar to SlidingPane1*/

            Calendar Calendar1 = new Calendar();

            Calendar1.ID = "Calendar1";

            Calendar1.Font.Size = 11;

            SlidingPane2.Controls.Add( Calendar1 );        

 

            /*Add Label control to SplitPane1*/

            Label Label1 = new Label();

            Label1.Text = "Sample Label";

            SplitPane2.Controls.Add( Label1 );

           

      }    

}
