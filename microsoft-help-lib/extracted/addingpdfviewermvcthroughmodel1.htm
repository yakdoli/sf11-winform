Adding Pdf Viewer MVC through Model

To add PDF viewer MVC through Model:

1.   Add the following references:

·      Syncfusion.Compression.Base

·      Syncfusion.Core

·      Syncfusion.Pdf.Base

·      Syncfusion.PdfViewer.Mvc

·      Syncfusion.Shared.Mvc

 

2.   Add the following code in Controller:

 

[Controller]

public class HomeController : Controller

    {      

 

        publicActionResult Index()

        {

            var model= newPdfViewerModel();          

            model.Load="Pdf File Path";

            ViewData["myModel"]=model;

            return View();

        }

 

        [AcceptVerbs(HttpVerbs.Post)]

        publicActionResult Index(PdfViewerParams args)

        {

            var model= newPdfViewerModel();          

            model.Load="Pdf File Path";           

            return model.PdfViewerHtmlActionResult(args);

        }

    }

 

 

 

3.   Add the following code in View:

 

[ASPX]

 

<% @   Page   Language ="C#"   MasterPageFile ="~/Views/Shared/Site.Master"   Inherits ="System.Web.Mvc.ViewPage"   %>

 

< asp : Content   ID ="Content1"   ContentPlaceHolderID ="TitleContent"   runat ="server">

   PdfViewer Model

</ asp : Content >

 

< asp : Content   ID ="Content2"   ContentPlaceHolderID ="MainContent"   runat ="server">

 

<% --Rendering the PdfViewer Control-- %>

   

     <% =  Html.PdfViewer( "PdfViewerModel" ,( PdfViewerModel )ViewData[ "myModel" ])  %>

 

</ asp : Content >

 

 

For more information on viewing PDF files, refer to Viewing PDF Files.
