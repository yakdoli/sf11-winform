Creating a Chart Control

The following code snippet illustrates the creation of chart control in Windows Phone application.

[XAML]

 

<phone:PhoneApplicationPage

    x:Class="WindowsPhoneApplication1.MainPage"

    xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"

    xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"

    xmlns:phone="clr-namespace:Microsoft.Phone.Controls;assembly=Microsoft.Phone"

    xmlns:shell="clr-namespace:Microsoft.Phone.Shell;assembly=Microsoft.Phone"

    xmlns:d="http://schemas.microsoft.com/expression/blend/2008"

    xmlns:mc="http://schemas.openxmlformats.org/markup-compatibility/2006"

   xmlns:syncfusion="clr-namespace:Syncfusion.Phone.Chart;assembly=Syncfusion.Chart.Phone"

    xmlns:local="clr-namespace:WindowsPhoneApplication1"

    mc:Ignorable="d" d:DesignWidth="480" d:DesignHeight="768"

    FontFamily="{StaticResource PhoneFontFamilyNormal}"

    FontSize="{StaticResource PhoneFontSizeNormal}"

    Foreground="{StaticResource PhoneForegroundBrush}"

    SupportedOrientations="Portrait" Orientation="Portrait"

    shell:SystemTray.IsVisible="True">

 

    <phone:PhoneApplicationPage.Resources>

        <local:StockData x:Key="data1">

            <local:FinancialStock StockId="101" MaxStockPrice="300" />

            <local:FinancialStock StockId="102" MaxStockPrice="260" />

            <local:FinancialStock StockId="103" MaxStockPrice="450" />

            <local:FinancialStock StockId="104" MaxStockPrice="210" />

            <local:FinancialStock StockId="105" MaxStockPrice="250" />

            <local:FinancialStock StockId="106" MaxStockPrice="200" />

        </local:StockData>

    </phone:PhoneApplicationPage.Resources>

 

    <!--LayoutRoot is the root grid where all page content is placed-->

    <Grid x:Name="LayoutRoot" Background="Transparent">

        <syncfusion:Chart x:Name="chart" Margin="2" Background="LightBlue">

            <syncfusion:ChartArea Margin="4">

                <syncfusion:ChartArea.Header>

                    <TextBlock Text="Stock Price Analysis" HorizontalAlignment="Center" VerticalAlignment="Center" FontSize="13" FontWeight="Bold"/>

                </syncfusion:ChartArea.Header>

 

                <syncfusion:ChartArea.PrimaryAxis>

                    <syncfusion:ChartAxis LabelFontWeight="Bold" HidePartialLabel="True">

                        <syncfusion:ChartAxis.Header>

                            <TextBlock Text="Stock Id" FontWeight="Bold"/>

                        </syncfusion:ChartAxis.Header>

                    </syncfusion:ChartAxis>

                </syncfusion:ChartArea.PrimaryAxis>

 

                <syncfusion:ChartArea.SecondaryAxis>

                    <syncfusion:ChartAxis LabelFontWeight="Bold">

                        <syncfusion:ChartAxis.Header>

                            <TextBlock Text="Stock Price" FontWeight="Bold"/>

                        </syncfusion:ChartAxis.Header>

                    </syncfusion:ChartAxis>

                </syncfusion:ChartArea.SecondaryAxis>

 

                <syncfusion:ChartArea.Legends>

                    <syncfusion:ChartLegend IconVisibility="Visible"/>

                </syncfusion:ChartArea.Legends>

 

                <!--Initialize the Chart data by using the DataSource property and X and Y axis values are initialized by using BindingPathX and BindingPathsY property-->

 

                <syncfusion:ChartSeries Type="Column" Label="Stock 1" Stroke="Black" StrokeThickness="2" DataSource="{StaticResource data1}" BindingPathX="StockId" BindingPathsY="MaxStockPrice" />

            </syncfusion:ChartArea>

        </syncfusion:Chart>

 

 

 

    </Grid>

 

 

</phone:PhoneApplicationPage>
