IList Data Binding

 

Simple IList based objects can be easily bound to the chart. The following image illustrates the chart with series associated by using the IList Data Binding.

 

The following lines of code describe how to bind the IList based instances to the chart.

 

 

[XAML]

 

        <syncfusion:Chart x:Name="MyChart" BorderThickness="3" Background="{StaticResource chartbackground}">

            <syncfusion:ChartArea>

                <syncfusion:ChartArea.Header>

                    <TextBlock Text="Area Header" FontFamily="Myraid Pro" FontSize="21" Foreground="#7F0900" FontWeight="Bold" TextAlignment="Center"/>

                </syncfusion:ChartArea.Header>

 

                <syncfusion:ChartArea.Legends>

                    <syncfusion:ChartLegend IconVisibility="Visible"/>

                </syncfusion:ChartArea.Legends>

 

                <syncfusion:ChartArea.PrimaryAxis>

                    <syncfusion:ChartAxis Interval="1" IsAutoSetRange="True" LineStroke="#828181" LineStrokeThickness="1" GridLineStroke="#828181" GridLineStrokeThickness="0.75" SmallTicksStroke="Transparent" LabelFontFamily="Myraid Pro" LabelFontSize="13">

                        <syncfusion:ChartAxis.Header>

                            <TextBlock Text="X-Axis" FontSize="13" FontFamily="Myraid Pro" FontWeight="Bold" TextAlignment="Center"/>

                        </syncfusion:ChartAxis.Header>

                    </syncfusion:ChartAxis>

                </syncfusion:ChartArea.PrimaryAxis>

 

                <syncfusion:ChartArea.SecondaryAxis>

                    <syncfusion:ChartAxis IsAutoSetRange="True" LineStrokeThickness="1" LineStroke="#828181" GridLineStroke="#828181" GridLineStrokeThickness="0.75" SmallTicksStroke="Transparent" LabelFontFamily="Myraid Pro" LabelFontSize="13">

                        <syncfusion:ChartAxis.Header>

                            <TextBlock Text="Y-Axis" FontSize="13" Foreground="#424242" FontFamily="Myraid Pro" FontWeight="Bold" TextAlignment="Center"/>

                        </syncfusion:ChartAxis.Header>

                    </syncfusion:ChartAxis>

                </syncfusion:ChartArea.SecondaryAxis>

 

                <syncfusion:ChartSeries Type="Column" x:Name="series" Label="Legend Data" Foreground="White"  StrokeThickness="0" />

            </syncfusion:ChartArea>

        </syncfusion:Chart>

 

 

 

[C#]

 

public MainPage()

        {

            InitializeComponent();

            AddChartDataSource(MyChart.Areas[0].Series[0]);

        }

 

        public void AddChartDataSource(ChartSeries series)

        {

            //Initialize the Model object collection for chart control.

            IList<XYValue> datas = new List<XYValue>();

            Random rd = new Random();

            datas.Add(new XYValue(1, rd.Next(0, 40)));

            datas.Add(new XYValue(2, rd.Next(0, 40)));

            datas.Add(new XYValue(3, rd.Next(0, 40)));

            datas.Add(new XYValue(4, rd.Next(0, 40)));

            datas.Add(new XYValue(5, rd.Next(0, 40)));

            datas.Add(new XYValue(6, rd.Next(0, 40)));

            datas.Add(new XYValue(7, rd.Next(0, 40)));

 

            //To bind the Model object collection into the chart series.

            series.DataSource = datas;

 

            //Initialize the Property name in Model object that provide values to X-Axis.

            series.BindingPathX = "X";

 

            //Initialize the Property name in Model object that provide values to Y-Axis.

            series.BindingPathsY = new List<string>() { "Y" };

        }

 

        public class XYValue

        {

            public int X

            {

                get;

                set;

            }

 

            public int Y

            {

                get;

                set;

            }

 

            public XYValue(int x, int y)

            {

                this.X = x;

                this.Y = y;

            }

        }
