Using Builder

The following steps guide in applying the default skin for the nodes and connectors through Builder.

1.   In the view, invoke the Diagram helper with the control ID as the first argument. 

                         

View

 

<% Node node1 = new Node()

   {

       Name = "Node1",

       Label = "Node1",      

       Shape = Shapes.Rectangle      

  };

  Node node2 = new Node()

   {

       Name = "Node2",

       Label = "Node2",      

       Shape = Shapes.Rectangle      

 

  };

  Node node3 = new Node()

   {

       Name = "Node3",

       Label = "Node3",      

       Shape = Shapes.Rectangle      

  };

 

 LineConnector line1 = new LineConnector()

   {

       Name="Line1",

       HeadNode = node1,

       TailNode = node2,

       HeadDecoratorShape = DecoratorShape.Arrow,

       TailDecoratorShape = DecoratorShape.Arrow,

       ConnectorType = ConnectorType.Orthogonal,

       Label="Line1"

   };

   LineConnector line2 = new LineConnector()

   {

       Name = "Line2",

       HeadNode = node1,

       TailNode = node3,

       HeadDecoratorShape = DecoratorShape.Arrow,

       TailDecoratorShape = DecoratorShape.Arrow,

       ConnectorType = ConnectorType.Orthogonal,

       Label = "Line2"

   };

%> 

<%{

      Html.Syncfusion().Diagram("FlatDiagram")

          .Nodes(nodes => nodes.Add(node1).Add(node2).Add(node3))

          .Connectors(conn => conn.Add(line1).Add(line2))

          .Width(900)

          .Height(500)

           .DiagramMode(DiagramMode.SVG)

          .LayoutType(LayoutType.DirectedTreeLayout)

         .AutoFormat(DiagramSkins.Vista)

          .Render();

  }

%>

 

2.   Build and run the application.
