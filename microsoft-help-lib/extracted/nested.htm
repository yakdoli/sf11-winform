Nested

 

Nested Relation Hierarchy is mainly used to link two or more tables with their field of Relation, for example, the ID will be common for the Order table and the Shipping table.

 

·      The Nested Relations is given in code, using the GridRelationDescriptor.

·      Hierarchical display can be implemented by defining multiple Grid levels.

·      Click on the top-left "+/-" column to expand the hierarchical records.

·      QueryShowRelationDisplayFields event is raised for each foreign key relation and lets you control at runtime, if the related fields of the Child table should be added to the FieldDescriptor collection.

 

[C#]

 

if (!IsPostBack)

 {

            DataTable employeeTable = GetEmployee();

 

            DataTable orderTable = GetOrder();

 

            //Specifying relations in grouping engine.

            GridRelationDescriptor gridRelationDescriptor = new GridRelationDescriptor();

            gridRelationDescriptor.ChildTableName = "Orders";

            gridRelationDescriptor.RelationKind = RelationKind.RelatedMasterDetails;

            gridRelationDescriptor.RelationKeys.Add("EmployeeID", "EmployeeID");

 

            //Adding relation to parent table

            GridGroupingControl1.TableDescriptor.Relations.Add(gridRelationDescriptor);

 

            // Register DataTable with SourceListSet,, so that RelationDescriptor can resolve the name

            this.GridGroupingControl1.Engine.SourceListSet.Add("Employees", employeeTable);

            this.GridGroupingControl1.Engine.SourceListSet.Add("Orders", orderTable);

            this.GridGroupingControl1.TableDescriptor.Name = "Employees";

 

            this.GridGroupingControl1.DataMember = "DefaultView";

            this.GridGroupingControl1.DataSource = employeeTable;

 

        }

        if (!CheckBox1.Checked)

            ScriptManager1.RegisterPostBackControl(GridGroupingControl1);

 

        this.GridGroupingControl1.DataSourceControlRowUpdating += new GridDataSourceControlRowUpdateEventHandler(GridGroupingControl1_DataSourceControlRowUpdating);

 

 

    }

 

    void GridGroupingControl1_DataSourceControlRowUpdating(object sender, GridDataSourceControlRowUpdateEventArgs e)

    {

        e.Cancel = true;

    }

   

    private DataTable GetOrder()

    {

        DataTable dt = new DataTable();

        OleDbDataAdapter da = new OleDbDataAdapter(AccessDataSource2.SelectCommand.ToString(), AccessDataSource2.ConnectionString.ToString());

        da.Fill(dt);

        DataColumn[] key = new DataColumn[1];

        key[0] = dt.Columns["OrderID"];

        dt.PrimaryKey = key;

        return dt;

 

    }

 

    private DataTable GetEmployee()

    {

        DataTable dt = new DataTable();

        OleDbDataAdapter da = new OleDbDataAdapter(AccessDataSource1.SelectCommand.ToString(), AccessDataSource1.ConnectionString.ToString());

        da.Fill(dt);

        DataColumn[] key = new DataColumn[1];

        key[0] = dt.Columns["EmployeeID"];

        dt.PrimaryKey = key;

        return dt;

 

    }

}
