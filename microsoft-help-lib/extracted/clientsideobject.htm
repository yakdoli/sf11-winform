The following code example demonstrates how to change the Splitter size programmatically.

 

[aspx]

 

<cc1:Splitter ID="Splitter1" runat="server" Height="250px" Width="250px" ResizeMode="Proportional">

    <cc1:SplitPane ID="LeftPane" runat="server">

        <span>Right SplitPane</span>

    </cc1:SplitPane>

    <cc1:SplitterBar ID="SplitterBar1" runat="server" CollapseMode="both" />

    <cc1:SplitPane ID="RightPane" runat="server">

        <span>Right SplitPane</span>

    </cc1:SplitPane>

</cc1:Splitter>

<input type="button" onclick="ChangeSplitterWidth(20)" value="Increase width" />

<input type="button" onclick="ChangeSplitterWidth(-20)" value="Decrease width" />

<input type="button" onclick="ChangeSplitterHeight(20)" value="Increase height" />

<input type="button" onclick="ChangeSplitterHeight(-20)" value="Decrease height" />

 

[javascript]

 

function ChangeSplitterWidth( nDelta )

{

    var nOldWidth = _sfSplitter1.GetWidth();

    var nNewWidth = nOldWidth + nDelta;

    if( nNewWidth > 0 )

    {

        _sfSplitter1.SetWidth( nNewWidth );

    }

}

function ChangeSplitterHeight( nDelta )

{

    var nOldHeight = _sfSplitter1.GetHeight();

    var nNewHeight = nOldHeight + nDelta;

    if( nNewHeight > 0 )

    {

        _sfSplitter1.SetHeight( nNewHeight );

    }

}

The following code example demonstrates how to use SplitPane's resize client side events.

 

[aspx]

 

<cc1:Splitter ID="Splitter1" runat="server" Height="300px" Width="600px">

    <cc1:SplitPane ID="MainPane" runat="server" Width="500px" ClientSideOnAfterResize="OnAfterResize(this)">

        <div id="MiddleEl" style="width: 100px; height: 100px; background-color: red; position: absolute;

            top: 100px; left: 100px;">

            this element is always in middle</div>

    </cc1:SplitPane>

    <cc1:SplitterBar ID="SplitterBar1" runat="server" CollapseMode="both" />

    <cc1:SplitPane ID="EndPane" runat="server">

        <span>any SplitPane</span>

    </cc1:SplitPane>

</cc1:Splitter>

 

[javascript]

function OnAfterResize(EventData)

{

    var oMiddleEl = document.getElementById("MiddleEl");

    oMiddleEl.style.top = ( EventData.newHeight - oMiddleEl.offsetHeight )/ 2 + "px";

    oMiddleEl.style.left = ( EventData.newWidth - oMiddleEl.offsetWidth )/2 + "px";

}

The following example code demonstrates how to have non-resizable SlidingPanes.

 

[ASPX]

 

<cc1:SlidingZone ID="SlidingZone" runat="server">

            <cc1:SlidingPane ID="SlidingPane1" runat="server" Title="NonResizablePane" Width="200px" ClientSideOnBeforeResize="false">content of nonresizable sliding pane</cc1:SlidingPane>

            <cc1:SlidingPane ID="SlidingPane3" runat="server" Title="ResizablePane" Width="200px">content of resizable sliding pane</cc1:SlidingPane>

        </cc1:SlidingZone>
