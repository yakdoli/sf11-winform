Adding Toolbar to the Mobile MVC application

In the Getting Started section, we discussed how to create a Mobile MVC application and add Tools package to the application. This section guides you to add Toolbar control to an application.

1.   In View, Invoke the Toolbar helper with the Toolbar ID as the first argument.

[ASPX]

<% = Html.MobSyncfusion().Toolbar("CoreToolbar")

   .Items(items =>

   {

                   items.Add().Value("New").Text("New").ImageUrl("~/Content/Toolbar/Images/new.png");

                   items.Add().Value("Open").Text("Open").ImageUrl("~/Content/Toolbar/Images/open.png");

                   items.Add().Value("Save").Text("Save").ImageUrl("~/Content/Toolbar/Images/save.png");

                   items.Add().Value("Print").Text("Print").ImageUrl("~/Content/Toolbar/Images/print.png");

 

items.Add().IsSeparator(true);

                   items.Add().Value("Cut").Text("Cut").ImageUrl("~/Content/Toolbar/Images/Cut.png");

   

   })%>

 

 

[Razor]

@{

   Html.MobSyncfusion().Toolbar("CoreToolbar")

   .Items(items =>

   {

                items.Add().Value("New").Text("New").ImageUrl("~/Content/Toolbar/Images/new.png");

                items.Add().Value("Open").Text("Open").ImageUrl("~/Content/Toolbar/Images/open.png");

                items.Add().Value("Save").Text("Save").ImageUrl("~/Content/Toolbar/Images/save.png");

                items.Add().Value("Print").Text("Print").ImageUrl("~/Content/Toolbar/Images/print.png");

 

items.Add().IsSeparator(true);

                items.Add().Value("Cut").Text("Cut").ImageUrl("~/Content/Toolbar/Images/Cut.png");

   })

   .Render();}

 

 

2.   Run the application.
