Using Builder

The following steps explain how to handle node deletion through Builder.

1.   In the view, create an object for the Node class and set the AllowDelete property.

2.   Invoke the Diagram helper with the control ID and set the Nodes property.

 

View

 

<% Node node1 = new Node()

   {

       Name = "Node1",

       AllowDelete=true

  };

  Node node2 = new Node()

   {

       Name = "Node2",

       AllowDelete=true

  };

  Node node3 = new Node()

   {

       Name = "Node3",

       AllowDelete=true

  };

 

 LineConnector line1 = new LineConnector()

   {

       Name="line1",

       HeadNode = node1,

       TailNode = node2,

       HeadDecoratorShape = DecoratorShape.None,

       TailDecoratorShape = DecoratorShape.None,

       ConnectorType = ConnectorType.Orthogonal,

       LineColor = "black",

       LineWidth = 2,

       Label="Line1"

   };

   LineConnector line2 = new LineConnector()

   {

       Name = "line2",

       HeadNode = node1,

       TailNode = node3,

       HeadDecoratorShape = DecoratorShape.None,

       TailDecoratorShape = DecoratorShape.None,

       ConnectorType = ConnectorType.Orthogonal,

       LineColor = "black",

       LineWidth = 2,

       Label = "Line2"

   };

 

%>

 

<%{

      Html.Syncfusion().Diagram("FlatDiagram")

          .Nodes(nodes => nodes.Add(node1).Add(node2).Add(node3))

          .Connectors(conn => conn.Add(line1).Add(line2))

          .Width(900)

          .Height(500)

          .DiagramMode(DiagramMode.SVG)

          .LayoutType(LayoutType.DirectedTreeLayout)

          .Render();

  }

%>

 

Note: If you want to create the diagram in the Canvas mode, change the DiagramMode to Canvas. By default the diagram is rendered in the SVG mode.

 

3.   Build and run the application.
