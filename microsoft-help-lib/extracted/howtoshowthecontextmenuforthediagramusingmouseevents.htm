How to show the Context Menu for the diagram using Mouse events?

 

Context Menus can be displayed in the diagram model and nodes on right clicking the same at runtime. The below steps illustrate this.

 

1.   Identify whether a mouse click occurred on a node or the model area using the MouseDown and MouseUp events as follows.

 

[C#]

 

private void diagram1_MouseDown(object sender, MouseEventArgs e)

{

      if( this.diagram1.Controller.ActiveTool != null )

      {

            m_bShowMenu = !this.diagram1.Controller.ActiveTool.InAction;

      }

}

// Handle the diagram.Controller.MouseUp event and display a context menu depending on whether the

// mouse click occurred over a diagram node or the diagram model area.

private void diagram1_ControllerMouseUp(object sender, MouseEventArgs e)

{

      if(e.Button == MouseButtons.Right && m_bShowMenu)

      {

            if(this.diagram1.Controller.NodesHit.Count > 0)

                  this.ShowNodeContextMenu(e);

            else

                  this.ShowModelContextMenu(e);            

      }

 

      m_bShowMenu = true;

}

 

[VB.NET]

 

Private Sub diagram1_MouseDown(ByVal sender As Object, ByVal e As MouseEventArgs)

If Me.diagram1.Controller.ActiveTool IsNot Nothing Then

m_bShowMenu = Not Me.diagram1.Controller.ActiveTool.InAction

End If

End Sub

' Handle the diagram.Controller.MouseUp event and display a context menu depending on whether the

' mouse click occurred over a diagram node or the diagram model area.

Private Sub diagram1_ControllerMouseUp(ByVal sender As Object, ByVal e As MouseEventArgs)

If e.Button = Windows.Forms.MouseButtons.Right AndAlso m_bShowMenu Then

If Me.Diagram1.Controller.NodesHit.Count > 0 Then

Me.ShowNodeContextMenu(e)

Else

Me.ShowModelContextMenu(e)

End If

End If

m_bShowMenu = True

End Sub

 

2.   Add the System.Windows.Form.ContextMenu namespace to the form and bar items with the necessary functionality. In this example, 'Copy', 'Delete', 'Edit' and 'Node Information' menu items were added.

 

[C#]

 

this.copyItem.Click += new System.EventHandler(this.copyItem_Click);

this.deleteNode.Click += new System.EventHandler(this.deleteNode_Click);

this.editTextNode.Click += new System.EventHandler(this.editTextNode_Click);

this.nodeInformation.Click += new System.EventHandler(this.nodeInformation_Click);

 

private void copyNode_Click(object sender, System.EventArgs e)

{

      this.diagram1.Controller.Copy();         

}          

 

// Delete the node over which the mouse was right clicked.

private void deleteNode_Click(object sender, System.EventArgs e)

{

      this.diagram1. Controller.Delete();

}

 

// Use the TextTool to invoke a user-edit of the TextNode that was clicked.

private void editTextNode_Click(object sender, System.EventArgs e)

{

    Node txt = this.diagram1.Controller.GetNodeAtPoint(Geometry.ConvertPoint(this.diagram1.Controller.MouseLocation));

    if (txt != null)

      {

        TextNode txtnode = txt as TextNode;

 

           if( txtnode != null && diagram1.Controller.TextEditor.IsEditable( txtnode ))

            diagram1.Controller.TextEditor.BeginEdit( txtnode, false );

      }

}

 

// Get the top-level node that was clicked and display the node details.

private void nodeInformation_Click(object sender, System.EventArgs e)

{

    Node digNode = this.diagram1.Controller.GetNodeAtPoint(Geometry.ConvertPoint( this.diagram1.Controller.MouseLocation));

 

    if (digNode != null)

      {

            // Display node information for the top-level child node.

        MessageBox.Show(this, String.Concat("You clicked a ", digNode.ToString(), "!"), "Node Information");

      }                

}

 

[VB.NET]

 

AddHandler Me.copyItem.Click, AddressOf Me.copyItem_Click

AddHandler Me.deleteNode.Click, AddressOf Me.deleteNode_Click

AddHandler Me.editTextNode.Click, AddressOf Me.editTextNode_Click

AddHandler Me.nodeInformation.Click, AddressOf Me.nodeInformation_Click

 

Private Sub copyNode_Click(ByVal sender As Object, ByVal e As System.EventArgs)

Me.diagram1.Controller.Copy()

End Sub

 

    ' Delete the node over which the mouse was right clicked.

Private Sub deleteNode_Click(ByVal sender As Object, ByVal e As System.EventArgs)

Me.diagram1.Controller.Delete()

End Sub

 

    ' Use the TextTool to invoke a user-edit of the TextNode that was clicked.

Private Sub editTextNode_Click(ByVal sender As Object, ByVal e As System.EventArgs)

Dim txt As Syncfusion.Windows.Forms.Diagram.Node = Me.Diagram1.Controller.GetNodeAtPoint(Geometry.ConvertPoint(Me.Diagram1.Controller.MouseLocation))

        If txt IsNot Nothing Then

            Dim txtnode As Syncfusion.Windows.Forms.Diagram.TextNode = TryCast(txt, Syncfusion.Windows.Forms.Diagram.TextNode)

 

            If txtnode IsNot Nothing AndAlso diagram1.Controller.TextEditor.IsEditable(txtnode) Then

                diagram1.Controller.TextEditor.BeginEdit(txtnode, False)

            End If

        End If

End Sub

3.   Now at runtime, when you click the menu items the corresponding events will be handled.
