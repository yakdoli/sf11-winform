Builder

To create a Radar chart through Builder:

1.   In Controller, return view to the corresponding View page.

 

[C#]

        public ActionResult SimpleChart()

        {           

            return View();

        }

 

 

2.   In the View page, invoke the ChartBuilder by using the control ID as the first argument.

3.   Add the Series to the ChartModel and set the series type to Radar, and add the Points to the series and set the style.

4.   Set the ChartModel and ChartArea properties.

 

View[ASPX]

 

    <%=Html.Chart("chart_Model").Text("Organisation Budget").Series(series =>

        {

 

            series.Add()

                  .Name("Allocated Budget")

                  .Type(Syncfusion.Windows.Forms.Chart.ChartSeriesType.Radar)

                  .Points(points =>

                  {

                      points.Add(0, 40);

                      points.Add(1, 20);

                      points.Add(2, 33);

                      points.Add(3, 25);

                      points.Add(4, 60);

                      points.Add(5, 20);

                  })

                  .ConfigItems(configItems =>

                  {

                      configItems.RadarItem(item =>

                      {

                          item.Type(Syncfusion.Windows.Forms.Chart.ChartRadarDrawType.Area);

                      });

                  });

            series.Add()

                  .Name("Actual Spending")

                  .Type(Syncfusion.Windows.Forms.Chart.ChartSeriesType.Radar)

                  .Points(points =>

                  {

                      points.Add(0, 50);

                      points.Add(1, 22);

                      points.Add(2, 25);

                      points.Add(3, 20);

                      points.Add(4, 20);

                      points.Add(5, 45);

                  })                 

                  .ConfigItems(configItems =>

                  {

                      configItems.RadarItem(item =>

                      {

                          item.Type(Syncfusion.Windows.Forms.Chart.ChartRadarDrawType.Area);

                      });

                  });

                       

        }).BorderAppearance(border =>

        {

            border.SkinStyle(Syncfusion.Windows.Forms.Chart.ChartBorderSkinStyle.Pinned);

        })    .Size(new System.Drawing.Size(500, 400))

              .Skins(ChartModelSkins.Office2007Blue)

              .ChartSeriesSkins(ChartSeriesSkins.Analog)

              .Series3D(true)             

              .PrimaryXAxis(xaxis => {

                  xaxis.RangeType(Syncfusion.Windows.Forms.Chart.ChartAxisRangeType.Set)

                       .Range(range => {

                           range.Min(0).Max(6).Interval(1);

                       });

              })

              .PrimaryYAxis(yaxis =>

              {

                  yaxis.RangeType(Syncfusion.Windows.Forms.Chart.ChartAxisRangeType.Set)

                       .Range(range =>

                       {

                            range.Min(0).Max(60).Interval(10);

                       });

              })   

    %>
