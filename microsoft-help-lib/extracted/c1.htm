The following code specifies the language properties values for C#.   

 

[C#]

 

public CSharpLanguage(EditControl control)

            : base(control)

        {

            ResourceDictionary dictionary = new ResourceDictionary();

            dictionary.Source = new Uri("/Syncfusion.Edit.Wpf;component/Languages/LanguageResources.xaml", UriKind.Relative);

            IsSplitTextToWords = true;

            this.Lexem = dictionary["CSharpLexems"] as LexemCollection;

            this.Formats = dictionary["CSharpFormats"] as FormatsCollection;

            this.Name = "CSharp";

            this.FileExtension = ".cs";

            this.TextForeground = control.Foreground;

            this.BlockStart = "{";

            this.BlockEnd = "}";

            this.ApplyColoring = true;

            this.CaseSensitive = true;

            this.SupportsOutlining = true;

            this.SupportsIntellisense = true;

        }
