Enable or Disable Zooming

 

Zoom can be enabled or disabled by the EnableZoom property. If the EnableZoom property is set to true, the Map can be zoomed. If the EnableZoom property is set to false, the Map cannot be zoomed.

 

[XAML]

       <maps:MapControl EnableZoom="False" ZoomMode="SingleClick" MinZoom="1" MaxZoom="10x:Name="Map" LayeredContent="{Binding ElementName=shapeControl}">

                       <maps:MapControl.Layers>

                            <maps:Layers>

                                <maps:ShapeFileLayer Uri="WindowsPhoneApplication1.ShapeFiles.wv.shp" TranslateZoomFactor="1" x:Name="shapeControl"  EnableColorPalette="True">

                                </maps:ShapeFileLayer>

                              </maps:Layers>

                        </maps:MapControl.Layers>           

       </maps:MapControl>

 

 

[C#]

            MapControl map = new MapControl();

            ShapeFileLayer shapeLayer = new ShapeFileLayer();

            shapeLayer.Uri = "WindowsPhoneApplication1.ShapeFiles.wv.shp";

            map.Layers.Items.Add(shapeLayer);

            map.LayeredContent = shapeLayer;

            map.EnableZoom= false;

            map.MinZoom=1d;

            map.MaxZoom=10d

            LayoutRoot.Children.Add(map);
