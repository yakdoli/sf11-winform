Adding UploadBox control to MVC Tools

 

In the Getting Started section, we discussed how to create an MVC application and add Tools package to the application. This section guides you to add the Upload Box to an application.

1.   In View, invoke the UploadBox helper with the UploadBox id as the first argument followed by the UploadBox’s properties and client-side events.

 

[ASPX]

<%= Html.Syncfusion().UploadBox("upload")

    .AllowMultipleFiles(true)

    .AsyncUpload(

        a =>

            a.SaveAction("Save","uploadbox")

                .RemoveAction("Remove", "uploadbox")

            .AutoUpload(true)

             

         )

    .AutoFormat(Skins.Sandune)%>

 

[Razor]

@{Html.Syncfusion().UploadBox("upload")

 

    .AllowMultipleFiles(true)

    .AsyncUpload(

        a =>

            a.SaveAction("Save", "UploadBox")

            .RemoveAction("Remove", "UploadBox")

            .AutoUpload(true)

         )

    .AutoFormat(Skins.Sandune).Render();}

 

 

 

 

 

[Controller]

  public ActionResult Save(IEnumerable<HttpPostedFileBase> upload)

        {

            // "upload" gets the attached file documents in the page

            foreach (var file in upload)

            {

                // "file.FileName" will get the name of the file uploaded. Some browsers send file names with full path

                var fileName = Path.GetFileName(file.FileName);

                var destinationPath = Path.Combine(Server.MapPath("~/App_Data"), fileName);

               

                file.SaveAs(destinationPath);

            }

                // Return an empty string to signify success

            return Content("");

        }

 

 

 

 

     public ActionResult Remove(string[] fileNames)

        {

            foreach (var fullName in fileNames)

            {

                var fileName = Path.GetFileName(fullName);

                var physicalPath = Path.Combine(Server.MapPath("~/App_Data"), fileName);

                if (System.IO.File.Exists(physicalPath))

                {

                   

                    System.IO.File.Delete(physicalPath);

                }

            }

            // Return an empty string to signify success

            return Content("");

        }

 

 

 

2.   Run the application.
