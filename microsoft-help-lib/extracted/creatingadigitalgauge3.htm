Creating a Digital Gauge

Following step-by-step instructions illustrate create a Digital Gauge and add elements.

 

1.   Add a Digital Gauge control to the Panel by using the following code.

 

[XAML]

 

<syncfusion:DigitalGauge Name="digitalGauge" Width="400" Height="100"/>

 

[C#]

 

DigitalGauge digitalGauge1 = new DigitalGauge();

digitalGauge1.Width = 300;

digitalGauge1.Height = 70;

LayoutRoot.Children.Add(digitalGauge1);

 

2.   Customize the Digital Gauge by setting the following properties.

 

[XAML]

 

<syncfusion:DigitalGauge Name="digitalGauge" Width="400" Height="100" CharacterHeight="30" SegmentSpacing="0.9" HorizontalContentAlignment="Center" CharacterSpacing="4.5" CharacterCount="14" Value="SYNCFUSION" CharacterType="SegmentFourteen">

</syncfusion:DigitalGauge>

 

[C#]

 

DigitalGauge digitalGauge = new DigitalGauge();

digitalGauge.SecondFrameFillColor = new SolidColorBrush(Colors.Black);

digitalGauge.DimmedBrush = new SolidColorBrush(Colors.Gray);

digitalGauge.Value = "Syncfusion";

digitalGauge.Foreground = new SolidColorBrush(Colors.Red);

digitalGauge.Width = 300;

digitalGauge.Height = 70;

digitalGauge.DimmedBrush = new SolidColorBrush(Colors.LightGray);

digitalGauge.SegmentSpacing = 1;

digitalGauge.HorizontalContentAlignment = HorizontalAlignment.Center;

digitalGauge.SegmentWidth = 2;

digitalGauge.CharacterCount = 10;

digitalGauge.CharacterHeight = 30;

digitalGauge.CharacterSpacing = 4;

digitalGauge.CharacterType = CharacterType.SegmentFourteen;

Rectangle rect = new Rectangle();

rect.Fill = new SolidColorBrush(Colors.Blue);

digitalGauge.InnerFrameContent = rect;

LayoutRoot.Children.Add(digitalGauge);
