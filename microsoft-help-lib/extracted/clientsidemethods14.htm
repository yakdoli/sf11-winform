Client-Side Methods

 

The tree-view control supports a rich set of client-side methods to modify its behavior.

2.   In JavaScript, use the methods as seen in the following code.

 

[Javscript]

 

<script type="text/javascript">

        function Expand() {

            //create an instance of the tree-view client-side object

            var treeViewObj = $find("myTreeView");

            //Code to expand a single node

            treeViewObj.ExpandNode("Silverlight");

            //Code to expand all nodes

            treeViewObj.ExpandAll();

        }

        function Collapse() {

            //create an instance of the tree-view client-side object

            var treeViewObj = $find("myTreeView");

            //Code to collapse a single node

            treeViewObj.CollapseNode("Silverlight");

            //Code to collapse all nodes

            treeViewObj.CollapseAll();

        }

        function Editing() {

            //create an instance of the tree-view client-side object

            var treeViewObj = $find("myTreeView");

            //Code to enable editing

            treeViewObj.EnableAllowEdit();

            //Code to disable editiing

            treeViewObj.DisableAllowEdit();

        }

        function TreeViewLines() {

            //create an instance of the tree-view client-side object

            var treeViewObj = $find("myTreeView");

            //Code to display tree-view lines

            treeViewObj.ShowLine();

            //Code to hide tree-view lines

            treeViewObj.HideLine();

        } function SelectNode() {

            // Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to select node.

            treeViewObj.SelectNode("Silverlight");

        }

        function UnSelectNode() {

            // Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to UnSelect node.

            treeViewObj.UnSelectNode("Silverlight");

        } function EnableNode() {

            // Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to enable node.

            treeViewObj.EnableNode("Silverlight");

        }

        function DisableNode() {

            // Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to disable node.

            treeViewObj.DisableNode("Silverlight");

        } function AddNode() {

            //Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to add new node.

            treeViewObj.AddNode("AspNet");

        }

        function RemoveNode() {

            // Create an instance of the tree-view client-side object.

            var treeViewObj = $find("myTreeView");

            // Code to remove node.

            treeViewObj.RemoveNode();

        }

      </script>
