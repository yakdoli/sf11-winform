Customizing Style

         

This feature enables you to customize the appearance of the JumpList. You can set the style for the JumpButton and the GroupButton. You can also set custom template for the JumpList items.

 

The following code illustrates how to customize the style:

           

[XAML]

 

   <syncfusion:JumpList x:Name="list"  >    

                

                <syncfusion:JumpList.GroupProvider>

                    <syncfusion:DistinctValueGroupProvider PropertyValue="Company"/>

                </syncfusion:JumpList.GroupProvider>

 

                  <!--Setting custom style to JumpButton-->

                <syncfusion:JumpList.JumpButtonStyle>

                    <Style TargetType="Button">                        

                        <Setter Property="HorizontalAlignment" Value="Stretch"></Setter>

                        <Setter Property="Margin" Value="0,20,0,0"></Setter>

                    </Style>

                </syncfusion:JumpList.JumpButtonStyle>

 

 

                 <!--Setting custom style to Group Button-->

                <syncfusion:JumpList.GroupButtonStyle>

                    <Style TargetType="Button">

                        <Setter Property="Width" Value="200"/>

                        <Setter Property="Margin" Value="10"/>

                        <Setter Property="Padding" Value="0"/>

                    </Style>

                </syncfusion:JumpList.GroupButtonStyle>

 

                 <!--Setting custom Template to JumplistItem -->

                <syncfusion:JumpList.ItemTemplate>

                    <DataTemplate>

                        <Grid Margin="4" Width="460" Background="Gray">         

                            <Grid.RowDefinitions>

                                <RowDefinition  Height="25*"></RowDefinition>

                                <RowDefinition Height="25*"></RowDefinition>

                                <RowDefinition Height="25*"></RowDefinition>

                                <RowDefinition Height="25*"></RowDefinition>

                            </Grid.RowDefinitions>

                            <Grid.ColumnDefinitions>

                                <ColumnDefinition Width="40*"></ColumnDefinition>

                                <ColumnDefinition Width="60*"></ColumnDefinition>

                            </Grid.ColumnDefinitions>

                             <TextBlock Text="TITLE: " Grid.Row="0" HorizontalAlignment="Center" ></TextBlock>

                             <TextBlock Grid.Row="0"  Grid.Column="1" Text="{Binding Title}"

                             FontSize="{StaticResource PhoneFontSizeMedium}"/>

 

                            <TextBlock Text="ARTIST: " Grid.Row="1" HorizontalAlignment="Center"> </TextBlock>

                            <TextBlock  Grid.Row="1" Grid.Column="1" Text="{Binding Artist}"

                             FontSize="{StaticResource PhoneFontSizeMedium}"/>

 

                            <TextBlock Text="PRICE: " Grid.Row="2" HorizontalAlignment="Center"></TextBlock>

                            <TextBlock  Grid.Row="2" Grid.Column="1" Text="{Binding Price}"

                             FontSize="{StaticResource PhoneFontSizeMedium}"/>

 

                            <TextBlock Text="YEAR: " Grid.Row="3" HorizontalAlignment="Center"></TextBlock>

                            <TextBlock  Grid.Row="3"  Grid.Column="1" Text="{Binding Year}"

                             FontSize="{StaticResource PhoneFontSizeMedium}"/>

                        </Grid>

                    </DataTemplate>

                </syncfusion:JumpList.ItemTemplate>

            </syncfusion:JumpList>
