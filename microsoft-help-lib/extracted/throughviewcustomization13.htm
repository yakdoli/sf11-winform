Through View Customization

 

 Note:

Pointer and StateIndicator Value Binding:

You should bind the Pointer Value and State Indicator Value(both properties should have equal Value) manually to indicate the current state of the indicator.

 

View:

Step 1:

 

Add the below code in your aspx file.

 

View[ASPX]

<%@ Page Language="C#" MasterPageFile="~/Views/Shared/Site.Master" Inherits="System.Web.Mvc.ViewPage" %>

<asp:Content ID="Content1" ContentPlaceHolderID="TitleContent" runat="server">

    Linear Gauge

</asp:Content>

 

<asp:Content ID="Content2" ContentPlaceHolderID="MainContent" runat="server">

<%--Rendering the Linear Gauge--%>

    <%=Html.Syncfusion().LinearGauge("Gauge")

        .Height(420)

        .Width(130)

        .GaugeSkins(GaugeSkins.VS2010)

        .Orientation(GaugeOrientation.Vertical)

        .FrameType(LinearGaugeFrameType.CroppedRectangle)

            .Scales(scale =>

            {

                scale.Add()

                    .Maximum(100)

                    .Minimum(0)

                    .MajorIntervalValue(10)

                    .MinorIntervalValue(2)

                    .ScaleBarSize(24)

                    .ScaleDirection(ScaleDirection.CounterClockwise)

                    .BorderWidth(4)

                    .ScaleBarLength(290)

                     .Ticks(Tick =>

                        {

                            Tick.Add()

                                .Angle(180)

                                .TickHeight(10)

                                .TickPlacement(ScalePlacement.Outside)

                                .TickShape(TickShape.Rectangle)

                                .TickStyle(TickStyle.MajorTick)

                                .DistanceFromScale(-24)

                                .TickWidth(4);

                            Tick.Add()

                                .Angle(180)

                                .TickHeight(6)

                                .TickPlacement(ScalePlacement.Outside)

                                .TickShape(TickShape.Rectangle)

                                .TickStyle(TickStyle.MinorTick)

                                .DistanceFromScale(-24)

                                .TickWidth(4);

                        })

                        .Pointers(pointer =>

                            {

                                pointer.AddBarPointer()

                                    .Value(72)

                                    .Opacity(0.7)

                                    .BarStyle(BarStyle.Rectangle)

                                    .PointerWidth(18);

                                pointer.AddMarker()

                                    .Value(72)

                                    .PointerPlacement(ScalePlacement.Outside)

                                    .MarkerStyle(MarkerStyle.Diamond)

                                    .PointerWidth(12)

                                    .PointerLength(17);

                            })

                            .Ranges(range =>

                                {

                                    range.Add()

                                        .RangePosition(ScalePlacement.Outside)

                                        .StartValue(60).EndValue(90)

                                        .EndWidth(13)

                                        .BackgroundBrush(Brushes.Red)

                                        .Opacity(0.4)

                                        .BorderWidth(0)

                                        .StartWidth(0);

                                })

                   .Labels(label =>

                       {

                           label.Add()

                               .DistanceFromScale(5)

                               .FontSize(15)

                               .TickPlacement(ScalePlacement.Inside)

                               .TickStyle(TickStyle.MajorTick)

                               .Angle(0);

                       });

            })

            //Adding the State Indicator in Linear Gauge.

                .StateIndicators(Indicator =>

                {

                    Indicator.Add()

                        //Specifying the height and width for the state indicator.

                        .IndicatorHeight(15)

                        .IndicatorWidth(15)

                        //Setting the value property, which is used to bind with pointer value,

                        //to decide active and inactive states of the indicator.

                        .Value(72)

                        //Setting the style for the indicator.

                        .IndicatorStyle(IndicatorStyle.CircularLED)

                        .BackgroundBrush(Brushes.LightBlue)

                        //Setting the location of the indicator.

                        .Location(new System.Windows.Point(50,89))

                       //Adding state ranges in which active start value and Active End value can be specified. If the                       

                        // pointer value is between the state range value, then the indicator is in active state you can 

                       //customize the active state of the indicator  using its ActiveBackground and ActiveBorderBrush

                       // property.

            

                           .StateRanges(s_Range =>

                            {

                                s_Range.Add()

                                    .StateRange(new StateRange(60, 90));

                            })

                        //Setting the background brush for the indicator when it is in active state.

                        .ActiveBackGroundBrush(Brushes.Red);

                    })               

        %>    

</asp:Content>

        }

}

Step 4:

 

In the above codes, you have mentioned that the State Range is between (60-90) and the Pointer Value is 32. Hence the state Indicator is in Inactive state. Now change the Pointer value to 72 and bind it with state Indicator Value and then see the State Indicator in Active State. You can see the difference in the color of the Indicator.

 

Figure 108: State Indicator-Active state
